const { powerMonitor } = require("electron");

const IDLE_THRESHOLD_SECONDS = 300;
const CHECK_INTERVAL_MS = 2000;

let activeWin = null;

async function initActiveWin() {
  if (!activeWin) {
    try {
      activeWin = (await import("active-win")).default;
    } catch (err) {
      console.error("[EDEN] Failed to load active-win:", err);
    }
  }
}

function classifyActivity(appName, windowTitle) {
  const app = (appName || "").toLowerCase();
  const title = (windowTitle || "").toLowerCase();

  const productivityKeywords = ["code", "cursor", "terminal", "xcode", "figma", "notion", "word", "excel", "docs", "postman"];
  const leisureKeywords = ["steam", "spotify", "discord", "netflix", "twitch", "vlc", "game", "reddit"];
  const leisureSites = ["youtube", "twitch", "reddit", "netflix", "twitter", "x.com"];

  if (productivityKeywords.some(keyword => app.includes(keyword) || title.includes(keyword))) {
    return "productivity";
  }
  if (leisureKeywords.some(keyword => app.includes(keyword) || title.includes(keyword)) ||
      leisureSites.some(site => title.includes(site))) {
    return "leisure";
  }
  return "productivity"; // Default active category fallback
}

function createActivityMonitor(onStatusChange) {
  let intervalId = null;
  let lastState = null;

  async function checkActivity() {
    await initActiveWin();
    const idleSeconds = powerMonitor.getSystemIdleTime();
    const isSystemIdle = idleSeconds >= IDLE_THRESHOLD_SECONDS;

    let appName = "";
    let windowTitle = "";
    let category = "productivity";

    if (activeWin) {
      try {
        const winInfo = await activeWin();
        if (winInfo && winInfo.owner) {
          appName = winInfo.owner.name;
          windowTitle = winInfo.title || "";
          category = classifyActivity(appName, windowTitle);
        }
      } catch (err) {
        // Handle window polling errors silently
      }
    }

    const newStatus = isSystemIdle ? "idle" : "active";
    
    // Trigger update if status or active app/title context changes
    if (
      !lastState ||
      lastState.status !== newStatus ||
      lastState.windowTitle !== windowTitle ||
      lastState.appName !== appName
    ) {
      lastState = {
        status: newStatus,
        category: isSystemIdle ? "idle" : category,
        idleSeconds,
        appName,
        windowTitle,
        timestamp: Date.now(),
      };

      onStatusChange(lastState);
    }
  }

  return {
    start() {
      if (intervalId !== null) return;
      checkActivity();
      intervalId = setInterval(checkActivity, CHECK_INTERVAL_MS);
    },
    stop() {
      if (intervalId !== null) {
        clearInterval(intervalId);
        intervalId = null;
      }
    },
    getStatus() {
      return lastState;
    },
  };
}

  createActivityMonitor,
  IDLE_THRESHOLD_SECONDS,

const freshWorld = (at = Date.now()): WorldState => ({
  ...structuredClone(DEFAULT_WORLD_STATE),

  waterSeed: Math.floor(Math.random() * 4294967295) + 1,

const getNonIdleCategory = (
  category: ActivityCategory,
  fallback: NonIdleCategory = "productivity"

  category === "productivity" || category === "leisure"

    : fallback;

const createId = (type: string) =>
  `${type}-${globalThis.crypto?.randomUUID?.() ?? `${Date.now()}-${Math.random().toString(36).slice(2)}`}`;

function createBuilding(world: WorldState, at: number): Building | null {
  if (world.buildings.length >= MAX_BUILDINGS) return null;
  const residents = residentOccupiedPlots(world.residents);
  // Let the growing city feel cottage-scale and garden-led while keeping every
  // existing building type available to the simulation.
  const catalog: (typeof BUILDING_CATALOG)[number][] = [];
  for (const entry of BUILDING_CATALOG) {
    const copies = entry.id === "solar-house" ? 4 : entry.id === "greenhouse" ? 2 : 1;
    for (let copy = 0; copy < copies; copy++) catalog.push(entry);
  }
  for (let attempt = 0; attempt < catalog.length; attempt++) {
    const config = catalog[Math.floor(Math.random() * catalog.length)]!;
    const plot = findFreePlot([...world.buildings, ...residents], world.plants, config.width, config.depth);
    if (!plot) continue;
    return { type: "building", id: createId("building"), name: config.name, variantId: config.id, gridPosition: plot, footprint: { width: config.width, depth: config.depth }, rotation: 0, floors: Math.min(config.floors, MAX_BUILDING_FLOORS), moss: 0, createdAt: at };
  }
  return null;
}

function createPlant(world: WorldState, at: number): Plant | null {
  if (world.plants.length >= MAX_PLANTS) return null;
  // Randomized candidates keep free navigation corridors around structures.
  const residents = residentOccupiedPlots(world.residents);
  const position = findFreePlantSite([...world.buildings, ...world.plants, ...residents]);
  if (!position) return null;
  const species = (["tree", "shrub", "fern", "flower", "grass"] as const)[Math.floor(Math.random() * 5)]!;
  return { type: "plant", id: createId("plant"), species, variantId: species, gridPosition: position, growth: 0.2 + Math.random() * 0.35, createdAt: at };
}


const EDEN_GROWTH_MAX = 3;

function edenRandom(id: string, salt = 0): number {
  let hash = 2166136261 ^ salt;
  for (let i = 0; i < id.length; i++) {
    hash = Math.imul(hash ^ id.charCodeAt(i), 16777619);
  }
  return (hash >>> 0) / 4294967295;
}

function growEden(world: WorldState, points: number): WorldState {
  if (points <= 0) return world;

  return {
    ...world,
    buildings: world.buildings.map(building => {
      const speed = 0.65 + edenRandom(building.id, 7) * 0.7;
      return {
        ...building,
        moss: Math.min(
          1,
          (building.moss ?? 0) + points * 0.025 * speed
        ),
      };
    }),
    plants: world.plants.map(plant => {
      const speed = 0.65 + edenRandom(plant.id, 13) * 0.7;
      const maxGrowth =
        1.4 + edenRandom(plant.id, 29) * 1.6;

      return {
        ...plant,
        growth: Math.min(
          Math.min(EDEN_GROWTH_MAX, maxGrowth),
          plant.growth + points * 0.075 * speed
        ),
      };
    }),
  };
}

function applyMilestones(worldInput: WorldState, at: number): WorldState {
  let world = worldInput;
  const consumedMilestones = { ...world.progression.consumedMilestones };
  for (const category of ["productivity", "leisure", "idle"] as const) {
    const total = world.activity[ACTIVITY_KEY[category]];
    const earned = Math.floor(total / ACTIVITY_MILESTONE_POINTS);
    let consumed = consumedMilestones[category];
    while (consumed < earned) {
      let changed = false;
      if (category === "productivity") {
        const eligible = world.buildings.find(building => (building.floors ?? 1) < MAX_BUILDING_FLOORS);
        if (eligible) {
          world = { ...world, buildings: world.buildings.map(building => building.id === eligible.id ? { ...building, floors: Math.min(MAX_BUILDING_FLOORS, (building.floors ?? 1) + 1), growth: 1 } : building) };
          changed = true;
        } else {
          const building = createBuilding(world, at);
          if (building) { world = { ...world, buildings: [...world.buildings, building] }; changed = true; }
        }
      } else if (category === "leisure") {
        const buildingCount = world.buildings.length;
        const treeCount = world.plants.filter(
          plant => plant.species === "tree"
        ).length;

        // Roughly 1 tree per building, with some variation.
        const maxTrees = Math.max(
          1,
          Math.ceil(buildingCount * 1.2)
        );

        // Limit total nature so buildings remain visually prominent.
        const maxNature = Math.max(
          2,
          Math.ceil(buildingCount * 1.8)
        );

        if (world.plants.length < maxNature) {
          let plant = createPlant(world, at);

          if (plant) {
            if (
              plant.species === "tree" &&
              treeCount >= maxTrees
            ) {
              const alternatives = [
                "shrub",
                "fern",
                "flower",
                "grass",
              ] as const;

              const species =
                alternatives[
                  Math.floor(Math.random() * alternatives.length)
                ]!;

              plant = {
                ...plant,
                species,
                variantId: species,
              };
            }

            world = {
              ...world,
              plants: [...world.plants, plant],
            };

            changed = true;
          }
        }

        // Existing buildings continue developing moss and vines.
        if (
          earned >= 3 &&
          world.buildings.some(
            building => (building.moss ?? 0) < 1
          )
        ) {
          world = {
            ...world,
            buildings: world.buildings.map(building => ({
              ...building,
              moss: Math.min(
                1,
                Math.max(
                  building.moss ?? 0,
                  (earned - 2) / 4
                )
              ),
            })),
          };

          changed = true;
        }

      } else if (world.residents.length < MAX_RESIDENTS) {
        const cell = findOpenResidentCell(world.buildings, world.plants, world.residents.map(resident => resident.position));
        if (!cell) break;
        const building = world.buildings[Math.floor(Math.random() * Math.max(1, world.buildings.length))];
        const resident: Resident = { type: "resident", id: createId("resident"), name: `Resident ${world.residents.length + 1}`, position: residentWorldPosition(cell), activity: "walking", homeId: building?.id, route: [], createdAt: at };
        world = { ...world, residents: [...world.residents, resident] }; changed = true;
      }
      if (!changed) break;
      consumed++;
    }
    consumedMilestones[category] = consumed;
  }
  return { ...world, progression: { ...world.progression, consumedMilestones } };
}

        ],

        ),

      if (world.cityDayKey !== dayKey) {

          if (isNight) {

          return { ...resident, activity: "walking", route, position: { ...resident.position, x: resident.position.x + dx / distance * step, z: resident.position.z + dz / distance * step } };
        }) };
      }

      world = growEden(applyMilestones(world, at), seconds / 60);

      return { world };
    });
  },

  // --------------------------------------------------

      world: {
        ...state.world,
        weather: structuredClone(weather),
      },
    })),

function roundedPerimeter(halfWidth: number, halfDepth: number, radius: number, y: number) {
  const points: Vector3[] = [];
  const corners = [
    { x: halfWidth - radius, z: -halfDepth + radius, from: -90 },
    { x: halfWidth - radius, z: halfDepth - radius, from: 0 },
    { x: -halfWidth + radius, z: halfDepth - radius, from: 90 },
    { x: -halfWidth + radius, z: -halfDepth + radius, from: 180 },
  ];
  for (const corner of corners) for (let step = 0; step <= 10; step++) {
    const angle = (corner.from + step * 9) * Math.PI / 180;
    points.push(new Vector3(corner.x + Math.cos(angle) * radius, y, corner.z + Math.sin(angle) * radius));
  }
  return points;
}

// Matching rounded corners for soil and glass.
// The glass bends smoothly underneath the terrarium.
const BASE_CURVE_RADIUS = 0.7;
const SOIL_CLEARANCE = 0.32;

function tankInsetAtHeight(height: number) {
  const h = Math.max(
    0,
    Math.min(BASE_CURVE_RADIUS, height)
  );
  const dy = BASE_CURVE_RADIUS - h;

  return BASE_CURVE_RADIUS - Math.sqrt(
    Math.max(0, BASE_CURVE_RADIUS ** 2 - dy ** 2)
  );
}

function createTankWallGeometry() {
  const positions: number[] = [];
  const indices: number[] = [];
  const contours: Vector3[][] = [];
  const bottomY = -TERRAIN_LIFT;

  // Generate a quarter-circle transition between the
  // horizontal glass floor and the vertical glass walls.
  for (let step = 0; step <= 16; step++) {
    const angle = step / 16 * Math.PI / 2;

    const height =
      BASE_CURVE_RADIUS * (1 - Math.cos(angle));

    const inset =
      BASE_CURVE_RADIUS * (1 - Math.sin(angle));

    contours.push(
      roundedPerimeter(
        TANK_WIDTH / 2 - inset,
        TANK_WIDTH / 2 - inset,
        TANK_RADIUS - inset,
        bottomY + height
      )
    );
  }

  // Continue vertically to the open top.
  contours.push(
    roundedPerimeter(
      TANK_WIDTH / 2,
      TANK_WIDTH / 2,
      TANK_RADIUS,
      TANK_HEIGHT
    )
  );

  const count = contours[0]!.length;

  for (const outline of contours) {
    if (outline.length !== count) {
      throw new Error("Inconsistent tank outline");
    }

    for (const point of outline) {
      positions.push(point.x, point.y, point.z);
    }
  }

  // Join neighboring contours into a continuous surface.
  for (let level = 0; level < contours.length - 1; level++) {
    for (let i = 0; i < count; i++) {
      const next = (i + 1) % count;

      const a = level * count + i;
      const b = level * count + next;
      const c = (level + 1) * count + i;
      const d = (level + 1) * count + next;

      indices.push(a, c, d, a, d, b);
    }
  }

  // Close the underside with transparent glass.
  // This also prevents an open hole when viewed from below.
  const center = positions.length / 3;
  positions.push(0, bottomY, 0);

  for (let i = 0; i < count; i++) {
    indices.push(center, i, (i + 1) % count);
  }

  const geometry = new BufferGeometry();

  geometry.setAttribute(
    "position",
    new Float32BufferAttribute(positions, 3)
  );

  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  geometry.computeBoundingSphere();

  return geometry;
}

function createRoundedGroundShape() {
  // The soil shares the exact corner centers of the glass.

  shape.moveTo(-half + radius, -half);

  shape.lineTo(half - radius, -half);
  shape.quadraticCurveTo(
    half, -half, half, -half + radius
  );

  shape.lineTo(half, half - radius);
  shape.quadraticCurveTo(
    half, half, half - radius, half
  );

  shape.lineTo(-half + radius, half);
  shape.quadraticCurveTo(
    -half, half, -half, half - radius
  );

  shape.lineTo(-half, -half + radius);
  shape.quadraticCurveTo(
    -half, -half, -half + radius, -half
  );

  return shape;
}

function waterRandom(seed: number) {
  let state = seed >>> 0;

  return () => {
    state = (Math.imul(state, 1664525) + 1013904223) >>> 0;
    return state / 4294967296;
  };
}

function WaterCourse({ seed }: { seed: number }) {
  const layout = useMemo(() => {
    const random = waterRandom(seed);
    const range = (min: number, max: number) =>
      min + random() * (max - min);

    const edge = BOARD_SIZE * 0.46;
    const riverPositions: number[] = [];

    // Randomized winding river.
    const startX = range(-BOARD_SIZE * 0.17, BOARD_SIZE * 0.17);
    const endX = range(-BOARD_SIZE * 0.17, BOARD_SIZE * 0.17);
    const bend1 = range(-BOARD_SIZE * 0.13, BOARD_SIZE * 0.13);
    const bend2 = range(-BOARD_SIZE * 0.07, BOARD_SIZE * 0.07);
    const phase = range(0, Math.PI * 2);
    const width = range(0.42, 0.78);

    const riverPoint = (t: number) => {
      const z = -edge + t * edge * 2;

      const rawX =
        startX * (1 - t) +
        endX * t +
        Math.sin(t * Math.PI * 2 + phase) * bend1 +
        Math.sin(t * Math.PI * 4 + phase * 0.6) * bend2;

      const w = width *
        (1 + 0.18 * Math.sin(t * Math.PI * 5 + phase));

      const x = Math.max(
        -edge + w,
        Math.min(edge - w, rawX)
      );

      return { x, z, w };
    };

    const steps = 100;

    for (let i = 0; i < steps; i++) {
      const a = riverPoint(i / steps);
      const b = riverPoint((i + 1) / steps);

      riverPositions.push(
        a.x - a.w, 0.065, a.z,
        b.x - b.w, 0.065, b.z,
        a.x + a.w, 0.065, a.z,

        a.x + a.w, 0.065, a.z,
        b.x - b.w, 0.065, b.z,
        b.x + b.w, 0.065, b.z
      );
    }

    const river = new BufferGeometry();
    river.setAttribute(
      "position",
      new Float32BufferAttribute(riverPositions, 3)
    );
    river.computeVertexNormals();

    // Organic shoreline geometry for lakes and ponds.
    const makeWaterBody = (
      x: number,
      z: number,
      radiusX: number,
      radiusZ: number
    ) => {
      const positions: number[] = [];
      const segments = 72;
      const phase1 = range(0, Math.PI * 2);
      const phase2 = range(0, Math.PI * 2);
      const wobble = range(0.06, 0.13);

      const point = (angle: number) => {
        const variation =
          1 +
          wobble * Math.sin(angle * 3 + phase1) +
          wobble * 0.45 * Math.cos(angle * 5 + phase2);

        return [
          x + Math.cos(angle) * radiusX * variation,
          0.078,
          z + Math.sin(angle) * radiusZ * variation
        ];
      };

      for (let i = 0; i < segments; i++) {
        const a = i / segments * Math.PI * 2;
        const b = (i + 1) / segments * Math.PI * 2;

        positions.push(
          x, 0.078, z,
          ...point(b),
          ...point(a)
        );
      }

      const geometry = new BufferGeometry();
      geometry.setAttribute(
        "position",
        new Float32BufferAttribute(positions, 3)
      );
      geometry.computeVertexNormals();

      return geometry;
    };

    // Put the lake and pond on opposite sides of the river.
    const lakeSide = random() < 0.5 ? -1 : 1;

    const lakeRadiusX = range(1.8, 3.1);
    const lakeRadiusZ = range(1.4, 2.5);

    const lakeX = lakeSide *
      range(BOARD_SIZE * 0.20, BOARD_SIZE * 0.29);
    const lakeZ = range(
      -BOARD_SIZE * 0.25,
      BOARD_SIZE * 0.25
    );

    const pondRadiusX = range(2.0, 3.8);
    const pondRadiusZ = range(1.7, 3.2);

    const pondX = -lakeSide *
      range(BOARD_SIZE * 0.20, BOARD_SIZE * 0.30);
    const pondZ = range(
      -BOARD_SIZE * 0.30,
      BOARD_SIZE * 0.30
    );


    const pond = makeWaterBody(
      pondX, pondZ, pondRadiusX, pondRadiusZ
    );

    return { river, pond };
  }, [seed]);

  useEffect(() => {
    return () => {
      layout.river.dispose();
      
      layout.pond.dispose();
    };
  }, [layout]);

  return (
    <group>
      <mesh geometry={layout.river} receiveShadow>
        <WaterSurface kind="river" />
      </mesh>

      

      <mesh geometry={layout.pond} receiveShadow>
        <WaterSurface kind="pond" />
      </mesh>
    </group>
  );
}

      return { x: 6 + Math.cos(waterAngle) * (3.7 + (index % 3) * 0.16), z: BOARD_SIZE * 0.29 + Math.sin(waterAngle) * (2.15 + (index % 4) * 0.13), sx: 0.62 + index % 3 * 0.12, sz: 0.57 + index % 4 * 0.1, color: index % 2 };
    }
    return { x: cosine * edge, z: sine * edge, sx: 0.74 + (index * 7 % 9) * 0.105, sz: 0.72 + (index * 5 % 8) * 0.11, color: index % 4 };
  }), []);
  useLayoutEffect(() => {
    const mesh = ref.current;
    if (!mesh) return;
    const matrix = new Matrix4();
    const colors = ["#47b85b", "#55c564", "#68ce69", "#3da857"];
    patches.forEach((patch, index) => {
      matrix.makeScale(patch.sx, 0.13 + index % 3 * 0.018, patch.sz).setPosition(patch.x, 0.075, patch.z);
      mesh.setMatrixAt(index, matrix);
      mesh.setColorAt(index, new Color(colors[patch.color]!));
    });
    mesh.count = patches.length;
    mesh.instanceMatrix.needsUpdate = true;
    if (mesh.instanceColor) mesh.instanceColor.needsUpdate = true;
    mesh.computeBoundingSphere();
    invalidate();
  }, [patches, invalidate]);
  return <instancedMesh ref={ref} args={[geometry, undefined, patches.length]} receiveShadow>
    <meshStandardMaterial color={night ? "#c0e4c9" : "#ffffff"} roughness={0.88} />
  </instancedMesh>;
}

/** A single draw batch for one geometry/material combination. */
const InstanceBatch = memo(function InstanceBatch({
  batch, night, owners, onSelectBuilding,
}: {
  batch: Batch;
  night: boolean;
  owners?: string[];
  onSelectBuilding?: (id: string) => void;
}) {
  const meshRef = useRef<InstancedMesh>(null);
  const invalidate = useThree(state => state.invalidate);
  // Spare capacity limits GPU buffer recreation as new objects are added.
  const capacity = Math.max(16, 2 ** Math.ceil(Math.log2(Math.max(1, batch.parts.length))));

  useLayoutEffect(() => {
    const mesh = meshRef.current;
    if (!mesh) return;
    const matrix = new Matrix4();
    const color = new Color();
    mesh.count = batch.parts.length;
    batch.parts.forEach((part, index) => {
      mesh.setMatrixAt(index, matrix.fromArray(part.matrix));
      mesh.setColorAt(index, color.set(part.color));
    });
    mesh.instanceMatrix.needsUpdate = true;
    if (mesh.instanceColor) mesh.instanceColor.needsUpdate = true;
    // Includes distant plots after additions, movement, or growth.
    mesh.computeBoundingSphere();
    mesh.computeBoundingBox();
    invalidate();
  }, [batch.parts, capacity, invalidate]);

  const glass = batch.finish === "glass";
  const greenhouse = batch.finish === "greenhouse";
  const solar = batch.finish === "solar";
  const metal = batch.finish === "metal";

  return (
    <instancedMesh
      ref={meshRef}
      args={[GEOMETRIES[batch.shape], undefined, capacity]}
      onClick={owners && onSelectBuilding ? event => {
        if (event.button !== 0 || event.delta > 5) return;
        const id = owners[event.instanceId ?? -1];
        if (!id) return;
        event.stopPropagation();
        onSelectBuilding(id);
      } : undefined}
      onPointerOver={owners ? () => { document.body.style.cursor = 'pointer'; } : undefined}
      onPointerOut={owners ? () => { document.body.style.cursor = ''; } : undefined}
      castShadow
      receiveShadow
    >
      <meshPhysicalMaterial
        color="#ffffff"
        transparent={glass || greenhouse}
        opacity={glass ? 0.4 : greenhouse ? 0.18 : 1}
        depthWrite={!glass && !greenhouse}
        side={2}
        roughness={glass || greenhouse ? 0.12 : solar ? 0.2 : metal ? 0.32 : 0.78}
        metalness={glass || greenhouse ? 0.04 : solar ? 0.3 : metal ? 0.36 : 0}
        emissive={glass ? "#fff0ab" : greenhouse ? "#8af5df" : "#000000"}
        emissiveIntensity={glass ? (night ? 0.58 : 0.16) : greenhouse ? (night ? 0.12 : 0.045) : 0}
        clearcoat={glass || greenhouse ? 0.7 : solar ? 0.9 : 0.12}
        clearcoatRoughness={glass || greenhouse ? 0.06 : solar ? 0.12 : 0.32}
        flatShading={!glass && !greenhouse && !metal}
      />
    </instancedMesh>
  );
});

function BatchLayer({ batches, night }: { batches: Batch[]; night: boolean }) {
  return <>{batches.map(batch => <InstanceBatch key={batch.key} batch={batch} night={night} />)}</>;
}

/** Store growth overrides age-based growth if the simulation supplies it. */
function BuildingsLayer({ night, onSelectBuilding }: { night: boolean; onSelectBuilding: (id: string) => void }) {
  const buildings = useWorldStore(selectBuildings);
  const [now, setNow] = useState(() => Date.now());

  useEffect(() => {
    const growing = buildings.filter(item => !hasExplicitGrowth(item) && buildingProgress(item, Date.now()) < 1);
    if (growing.length === 0) return;
    const timer = window.setInterval(() => {
      const next = Date.now();
      setNow(next);
      if (growing.every(item => buildingProgress(item, next) >= 1)) window.clearInterval(timer);
    }, GROWTH_REFRESH_MS);
    return () => window.clearInterval(timer);
  }, [buildings]);

  const { batches, ownerByBatch } = useMemo(() => {
    const parts: Part[] = [];
    const ownerByPart: string[] = [];
    for (const item of buildings) {
      const firstPart = parts.length;
      const at = cellPoint(
        item.gridPosition.x + (item.footprint.width - 1) / 2,
        item.gridPosition.z + (item.footprint.depth - 1) / 2,
      );
      // Footprints are occupied grid dimensions. A quarter-turn swaps the local
      // dimensions so a rectangular building still fits its recorded plot.
      const { width, depth } = buildingDimensions(item.footprint, item.rotation);
      const progress = buildingProgress(item, now);
      const stage = growthStage(progress);
      const model = buildingModel(item.variantId, width, depth, stage, hashId(item.id), item.floors, item.moss);
      // Horizontal size stays fixed so growth cannot consume neighboring plots.
      const buildingVariation = 0.75 + (hashId(item.id) % 100) / 100 * 0.8;
    const root = compose(
      [at.x, 0, at.z],
      [0, item.rotation, 0],
      [1, (0.65 + progress * 0.35) * buildingVariation, 1]
    );
      placeModel(model, root, parts);

      // Climbing vines and leaves appear as moss increases.
      const moss = Math.max(0, Math.min(1, item.moss ?? 0));

      if (moss > 0.05) {
        const height = (item.floors ?? 1) * 0.55 + 0.5;
        const vineCount = Math.ceil(moss * 12);

        for (let i = 0; i < vineCount; i++) {
          const side = i % 4;
          const fraction = ((i * 7) % 11) / 11 - 0.5;

          const localX =
            side < 2
              ? fraction * width * 0.85
              : (side === 2 ? -1 : 1) * width * 0.46;

          const localZ =
            side < 2
              ? (side === 0 ? -1 : 1) * depth * 0.46
              : fraction * depth * 0.85;

          const angle = item.rotation;

          const x =
            at.x +
            localX * Math.cos(angle) +
            localZ * Math.sin(angle);

          const z =
            at.z -
            localX * Math.sin(angle) +
            localZ * Math.cos(angle);

          const vineHeight =
            height * moss * (0.5 + (i % 4) * 0.12);

          parts.push({
            shape: "stem",
            finish: "matte",
            color: i % 3 === 0 ? "#397b3c" : "#4d9b4a",
            matrix: compose(
              [x, vineHeight / 2, z],
              [0, 0, 0],
              [
                0.035 + moss * 0.035,
                vineHeight,
                0.035 + moss * 0.035,
              ]
            ),
          });

          const leafCount = Math.ceil(moss * 6);

          for (let j = 0; j < leafCount; j++) {
            const leafY =
              ((j + 1) / (leafCount + 1)) * vineHeight;

            parts.push({
              shape: "leaf",
              finish: "matte",
              color: j % 2 === 0 ? "#56a94c" : "#7bc768",
              matrix: compose(
                [
                  x + (j % 2 === 0 ? -0.06 : 0.06),
                  leafY,
                  z,
                ],
                [0, 0, j % 2 === 0 ? -0.5 : 0.5],
                [
                  0.09 + moss * 0.06,
                  0.045 + moss * 0.035,
                  0.065,
                ]
              ),
            });
          }
        }
      }

      // Parts appended by this building (including moss/vines) share its ID.
      for (let i = firstPart; i < parts.length; i++) ownerByPart.push(item.id);
    }
    const ownerByBatch = new Map<string, string[]>();
    parts.forEach((part, index) => {
      const key = `${part.shape}:${part.finish}`;
      let list = ownerByBatch.get(key);
      if (!list) { list = []; ownerByBatch.set(key, list); }
      list.push(ownerByPart[index]!);
    });
    return { batches: batchParts(parts), ownerByBatch };
  }, [buildings, now]);

  return <>{batches.map(batch => <InstanceBatch
    key={batch.key}
    batch={batch}
    night={night}
    owners={ownerByBatch.get(batch.key)}
    onSelectBuilding={onSelectBuilding}
  />)}</>;
}

function PlantsLayer({ night }: { night: boolean }) {
  const plants = useWorldStore(selectPlants);
  const batches = useMemo(() => {
    const parts: Part[] = [];
    for (const item of plants) {
      const at = cellPoint(item.gridPosition.x, item.gridPosition.z);
      const seed = hashId(item.id);
      const speciesScale = item.species === "tree" ? 2.2 : item.species === "shrub" ? 1.6 : 1.35;
      const growth = Math.max(0, Math.min(3, item.growth));

    const natureMultiplier =
      item.species === "tree" ? 1.65 :
      item.species === "shrub" ? 1.4 :
      item.species === "fern" ? 1.35 :
      item.species === "flower" ? 1.5 :
      1.25;

    const individualSize = 0.7 + (seed % 100) / 100 * 0.9;

    const scale =
      (0.25 + growth * 0.65) *
      individualSize *
      (CELL_SIZE / 1.5) *
      speciesScale *
      natureMultiplier;
      const root = compose([at.x, 0, at.z], [0, seed % 360 * Math.PI / 180, 0], [scale, scale, scale]);
      placeModel(plantModel(item.species, seed), root, parts);
    }
    return batchParts(parts);
  }, [plants]);
  return <BatchLayer batches={batches} night={night} />;
}

function ResidentsLayer({ night }: { night: boolean }) {
  const residents = useWorldStore(selectResidents);
  const batches = useMemo(() => {
    const parts: Part[] = [];
    for (const item of residents) {
      if (item.activity === "resting") continue;
      const at = residentPoint(item.position.x, item.position.z);
      parts.push({ shape: "capsule", finish: "matte", color: "#9b9d98", matrix: compose([at.x, 0.13, at.z]) });
    }
    return batchParts(parts);
  }, [residents]);
  return <BatchLayer batches={batches} night={night} />;
}

function useNight() {
  const simulationTime = useWorldStore(state => state.world.simulationTime);
  return getDayPhase(simulationTime) === "night";
}

function App() {
    const setSystemIdle = useWorldStore(
    (state) => state.setSystemIdle
  );
    // Simulation timer and activity tracking
  const advanceTime = useWorldStore(state => state.advanceTime);
  const recordInput = useWorldStore(state => state.recordInput);
  const setActivityCategory = useWorldStore(state => state.setActivityCategory);
  const addBuilding = useWorldStore(state => state.addBuilding);
  const addPlant = useWorldStore(state => state.addPlant);
  const resetWorld = useWorldStore(state => state.resetWorld);

useEffect(() => {
    const api = window.edenAPI;

    if (!api) return;

    let mounted = true;
    let latestTimestamp = -Infinity;

    const handleActivity = (activity: {
      status: "active" | "idle";
      category?: "productivity" | "leisure" | "idle";
      timestamp: number;
    }) => {
      if (!mounted || activity.timestamp < latestTimestamp) return;

      latestTimestamp = activity.timestamp;
      setSystemIdle(activity.status === "idle");

      if (activity.category && activity.status !== "idle") {
        setActivityCategory(activity.category);
      }
    };

    const unsubscribe = api.onActivityChanged(handleActivity);

    api.getCurrentActivity()
      .then((activity) => {
        if (activity) handleActivity(activity);
      })
      .catch(console.error);

    return () => {
      mounted = false;
      unsubscribe();
    };
  }, [setSystemIdle, setActivityCategory]); // <-- Include both here


  // Primitive subscriptions keep the toolbar out of world movement/growth updates.
  const buildingCount = useWorldStore(state => selectBuildings(state).length);
  const plantCount = useWorldStore(state => selectPlants(state).length);
  const worldActivity = useWorldStore(selectActivity);
  const weather = useWorldStore(state => state.world.weather);
  const simulationTime = useWorldStore(state => state.world.simulationTime);
  const [variant, setVariant] = useState<BuildingVariant>("solar-house");
  const [plantType, setPlantType] = useState<Plant["species"]>("tree");
  const [panelOpen, setPanelOpen] = useState(false);
  const [windowHovered, setWindowHovered] = useState(false);
  const [showGrid, setShowGrid] = useState(false);
  const [shadows, setShadows] = useState(true);
  const [notice, setNotice] = useState("");
  const [selectedBuildingId, setSelectedBuildingId] = useState<string | null>(null);

  useEffect(() => {
    let previous = Date.now();
    let lastInput = 0;
    const timer = window.setInterval(() => {
      const now = Date.now();
      advanceTime(Math.max(0, now - previous) / 1_000, now);
      previous = now;
    }, 1_000);
    const onInput = (event: Event) => {
      const target = event.target;
      const editing = target instanceof HTMLElement && (target.isContentEditable || !!target.closest("input, textarea, select"));
      if (!editing && event instanceof KeyboardEvent && event.altKey && ["Digit1", "Digit2", "Digit3"].includes(event.code)) {
        event.preventDefault();
        recordInput();
        setActivityCategory(event.code === "Digit1" ? "productivity" : event.code === "Digit2" ? "leisure" : "idle");
        return;
      }
      // Wheel events can arrive dozens of times per second.
      const now = Date.now();
      // Only track window input when running in a browser.
      // Electron already tracks activity across the entire Mac.
      if (!window.edenAPI) {
        if (now - lastInput >= 200) { recordInput(); lastInput = now; }
      }
    };
    window.addEventListener("pointerdown", onInput, { passive: true });
    window.addEventListener("keydown", onInput);
    window.addEventListener("wheel", onInput, { passive: true });
    return () => {
      window.clearInterval(timer);
      window.removeEventListener("pointerdown", onInput);
      window.removeEventListener("keydown", onInput);
      window.removeEventListener("wheel", onInput);
    };
  }, [advanceTime, recordInput, setActivityCategory]);
  
  function handleAddBuilding() {
    const config = BUILDING_CATALOG.find(entry => entry.id === variant)!;
    // Read current state at click time, including rapid successive additions.
    const state = useWorldStore.getState();

    const item: Building = {

      gridPosition: plot, footprint: { width: config.width, depth: config.depth },
      rotation: 0, createdAt: Date.now(),
    };
    try {
      addBuilding(item);
      const accepted = selectBuildings(useWorldStore.getState()).some(building => building.id === item.id);
      setNotice(accepted ? `${config.name} added. It grows over ${BUILDING_GROWTH_SECONDS / 60} minutes.` : "The store rejected this building. Update its grid limits and variant catalog.");
    } catch (error) {
      setNotice(error instanceof Error ? error.message : "The building could not be added.");
    }
  }

  function handleAddPlant() {
    const state = useWorldStore.getState();
    const residents = residentOccupiedPlots(selectResidents(state));
    const position = findFreePlantSite([...selectBuildings(state), ...selectPlants(state), ...residents]);
    if (!position) { setNotice("No suitable planting space remains."); return; }
    addPlant({ type: "plant", id: createRendererId("plant"), species: plantType, variantId: plantType, gridPosition: position, growth: 0.9, createdAt: Date.now() });
    setNotice(`${plantType} added.`);
  }

    <main
      className="app"
      style={{ position: "relative", width: "100vw", height: "100vh" }}
      onPointerEnter={() => setWindowHovered(true)}
      onPointerLeave={() => setWindowHovered(false)}

        gl={GL_OPTIONS}

        ...PANEL_STYLE,

        overflowY: "auto",
        ...(panelOpen ? {} : { width: "auto", padding: 7 }),
      }}>
        <button type="button" style={CONTROL_STYLE} onClick={() => setPanelOpen(open => !open)} aria-expanded={panelOpen} aria-controls="eden-building-panel">
          {panelOpen ? "Close controls" : "Build"}
        </button>
        {panelOpen && <div id="eden-building-panel" style={{ display: "flex", flexDirection: "column", gap: 8 }}>
          <strong>{GRID_SIZE}×{GRID_SIZE} map · {buildingCount} buildings · {plantCount} plants</strong>
          <small>Productivity {worldActivity.productivityPoints.toFixed(1)} · Leisure {worldActivity.leisurePoints.toFixed(1)} · Idle {worldActivity.idlePoints.toFixed(1)} points</small>
          <small>{weather.condition} · {getDayPhase(simulationTime)} · 10 minute cycle</small>
          <label htmlFor="eden-building-type">Building type</label>
          <select id="eden-building-type" value={variant} onChange={event => setVariant(event.target.value as BuildingVariant)} style={CONTROL_STYLE}>
            {BUILDING_CATALOG.map(entry => <option key={entry.id} value={entry.id}>{entry.name} · {entry.width}×{entry.depth}</option>)}
          </select>
          <button type="button" onClick={handleAddBuilding} style={CONTROL_STYLE}>Add building</button>
          <label htmlFor="eden-plant-type">Plant type</label>
          <select id="eden-plant-type" value={plantType} onChange={event => setPlantType(event.target.value as Plant["species"])} style={CONTROL_STYLE}>

          <small>New buildings grow through starter, expanded, and mature stages.</small>
          <span role="status" style={{ lineHeight: 1.4 }}>{notice}</span>
        </div>}
      </div>
      <div className="wordmark">EDEN<span>.EXE</span></div>
      <BuildingInsights buildingId={selectedBuildingId} onClose={() => setSelectedBuildingId(null)} />

function pointInside(bounds, point) {
  return (
    point.x >= bounds.x &&
    point.x < bounds.x + bounds.width &&
    point.y >= bounds.y &&
    point.y < bounds.y + bounds.height
  );

function stopHoverMonitoring() {

    hoverTimer = null;
  }

  if (hideTimer) {

  }

function startHoverMonitoring() {
  stopHoverMonitoring();

  hoverTimer = setInterval(() => {
    if (!mainWindow || mainWindow.isDestroyed()) {
      return;
    }

    if (!titlebarWindow || titlebarWindow.isDestroyed()) {
      return;
    }

    if (
      !mainWindow.isVisible() ||
      mainWindow.isMinimized() ||
      mainWindow.isFullScreen()
    ) {
      titlebarWindow.hide();
      return;
    }

    const cursor = screen.getCursorScreenPoint();

    const overMain = pointInside(
      mainWindow.getBounds(),
      cursor,
    );

    const overTitlebar =
      titlebarWindow.isVisible() &&
      pointInside(titlebarWindow.getBounds(), cursor);

    if (overMain || overTitlebar) {
      if (hideTimer) {
        clearTimeout(hideTimer);
        hideTimer = null;
      }

      if (!titlebarWindow.isVisible()) {
        if (syncTitlebarPosition()) {
          titlebarWindow.showInactive();
        }
      }
    } else if (
      titlebarWindow.isVisible() &&
      !hideTimer
    ) {
      hideTimer = setTimeout(() => {
        hideTimer = null;

        if (
          titlebarWindow &&
          !titlebarWindow.isDestroyed()
        ) {
          titlebarWindow.hide();
        }
      }, HIDE_DELAY);
    }
  }, HOVER_INTERVAL);
}

import { DEFAULT_WORLD_STATE, getCityDayKey } from "../simulation/worldDefaults";
import { MAX_BUILDING_FLOORS } from "../edenModels";
import type { ActivityCategory, Building, Plant, Resident, WorldState } from "../simulation/types";
import { useWorldStore } from "./worldStore";

const BROWSER_SAVE_KEY = "eden-world-v1";
let initialized = false;

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null;
}

/** Normalize worlds saved by either side of the earlier merge. */
function normalizeWorld(input: unknown): WorldState | null {
  if (!isRecord(input)) return null;
  const source = isRecord(input.world) ? input.world : input;
  if (
    !Array.isArray(source.buildings) ||
    !Array.isArray(source.plants) ||
    !Array.isArray(source.residents)
  )
    return null;

  const activitySource = isRecord(source.activity) ? source.activity : {};
  const legacyCategory =
    typeof source.activity === "string" ? source.activity : activitySource.category;
  const category: ActivityCategory =
    legacyCategory === "productivity" || legacyCategory === "leisure" || legacyCategory === "idle"
      ? legacyCategory
      : "idle";
  const buildings = source.buildings.map((value, index) => {
    const building = isRecord(value) ? value : {};
    return {
      ...building,
      type: "building",
      id: typeof building.id === "string" ? building.id : `building-${index + 1}`,
      name: typeof building.name === "string" ? building.name : "Solar Workshop",
      variantId: typeof building.variantId === "string" ? building.variantId : "solar-workshop",
      floors: typeof building.floors === "number" && Number.isFinite(building.floors)
        ? Math.max(1, Math.min(MAX_BUILDING_FLOORS, Math.floor(building.floors)))
        : undefined,
      moss: typeof building.moss === "number" && Number.isFinite(building.moss)
        ? Math.max(0, Math.min(1, building.moss))
        : 0,
    };
  }) as Building[];
  const plants = source.plants.map((value, index) => {
    const plant = isRecord(value) ? value : {};
    const species = typeof plant.species === "string" ? plant.species : "broadleaf";
    return {
      ...plant,
      type: "plant",
      id: typeof plant.id === "string" ? plant.id : `plant-${index + 1}`,
      species,
      variantId: typeof plant.variantId === "string" ? plant.variantId : species,
    };
  }) as Plant[];
  const residents = source.residents.map((value, index) => {
    const resident = isRecord(value) ? value : {};
    return {
      ...resident,
      type: "resident",
      id: typeof resident.id === "string" ? resident.id : `resident-${index + 1}`,
    };
  }) as Resident[];

  const grid = isRecord(source.grid) ? source.grid : {};
  const weather = isRecord(source.weather) ? source.weather : DEFAULT_WORLD_STATE.weather;
  const progression = isRecord(source.progression) ? source.progression : {};
  return {
    grid: { ...DEFAULT_WORLD_STATE.grid, ...grid, width: 20, depth: 20, cellSize: 1.5 } as WorldState["grid"],
    buildings,
    plants,
    residents,
    weather: {
      condition: weather.condition === "cloudy" || weather.condition === "rainy" ? weather.condition : "sunny",
      updatedAt: typeof weather.updatedAt === "number" ? weather.updatedAt : Date.now(),
    },
    simulationTime: typeof source.simulationTime === "number" ? source.simulationTime : 0,
    activity: {
      category,
      productivityPoints:
        typeof activitySource.productivityPoints === "number"
          ? activitySource.productivityPoints
          : isRecord(source.progression) &&
              typeof source.progression.productivitySeconds === "number"
            ? source.progression.productivitySeconds
            : 0,
      leisurePoints:
        typeof activitySource.leisurePoints === "number"
          ? activitySource.leisurePoints
          : isRecord(source.progression) && typeof source.progression.leisureSeconds === "number"
            ? source.progression.leisureSeconds
            : 0,
      idlePoints:
        typeof activitySource.idlePoints === "number"
          ? activitySource.idlePoints
          : isRecord(source.progression) && typeof source.progression.idleSeconds === "number"
            ? source.progression.idleSeconds
            : 0,
    },
    lastActiveAt: typeof source.lastActiveAt === "number" ? source.lastActiveAt : Date.now(),
    progression: {
      unlockedBuildings: Array.isArray(progression.unlockedBuildings)
        ? progression.unlockedBuildings.filter((item): item is string => typeof item === "string")
        : [],
      consumedMilestones: isRecord(progression.consumedMilestones)
        ? {
            productivity: typeof progression.consumedMilestones.productivity === "number" ? Math.max(0, Math.floor(progression.consumedMilestones.productivity)) : 0,
            leisure: typeof progression.consumedMilestones.leisure === "number" ? Math.max(0, Math.floor(progression.consumedMilestones.leisure)) : 0,
            idle: typeof progression.consumedMilestones.idle === "number" ? Math.max(0, Math.floor(progression.consumedMilestones.idle)) : 0,
          }
        : { productivity: 0, leisure: 0, idle: 0 },
    },
    cityDayKey: typeof source.cityDayKey === "string"
      ? source.cityDayKey
      : getCityDayKey(new Date(typeof source.lastActiveAt === "number" ? source.lastActiveAt : Date.now())),
    cycleStartedAt: typeof source.cycleStartedAt === "number" ? source.cycleStartedAt : Date.now(),
  };
}

export async function initializeWorldPersistence(): Promise<void> {
  if (initialized) return;

  const api = window.edenAPI;
  const useBrowserStorage = !api;
  try {
    const savedValue = api ? await api.loadWorld() : localStorage.getItem(BROWSER_SAVE_KEY);
    if (savedValue) {
      const parsed =
        typeof savedValue === "string" ? (JSON.parse(savedValue) as unknown) : savedValue;
      const world = normalizeWorld(parsed);
      if (!world) throw new Error("Saved world has an unsupported format.");
      if (world.cityDayKey !== getCityDayKey()) useWorldStore.getState().resetWorld();
      else useWorldStore.getState().loadWorld(world);
    }
  } catch (error) {
    console.error("Failed to load saved world:", error);
    // Don't overwrite a save that failed to load or validate.
    return;
  }

  let saveQueue = Promise.resolve();
  useWorldStore.subscribe((state, previousState) => {
    if (state.world === previousState.world) return;
    const snapshot = structuredClone(state.world);
    saveQueue = saveQueue
      .catch(() => undefined)
      .then(async () => {
        if (api) await api.saveWorld(snapshot);
        else if (useBrowserStorage)
          localStorage.setItem(BROWSER_SAVE_KEY, JSON.stringify(snapshot));
      })
      .catch((error) => console.error("Failed to save world:", error));
  });

  initialized = true;
}

  MAX_BUILDING_FLOORS,
  buildingProgress,
  growthStage,

  selectActivity,

// Mirrors the displayed building floor calculation used by the renderer.
function displayedFloors(building: Building, now: number): number {
  const config = BUILDING_CATALOG.find(item => item.id === building.variantId);
  const stage = growthStage(buildingProgress(building, now));
  const fallback = 1 + ((config?.floors ?? 1) - 1) * stage / 2;
  return Math.min(MAX_BUILDING_FLOORS, Math.max(1, Math.round(building.floors ?? fallback)));
}

// The city's statistics reset at local 4 AM; unlike a fixed 24h calculation,
// calculating consecutive local 4 AM boundaries also handles DST changes.
function cityDayPercentage(timestamp: number): number {
  const start = new Date(timestamp);
  start.setHours(4, 0, 0, 0);
  if (timestamp < start.getTime()) start.setDate(start.getDate() - 1);

  const end = new Date(start);
  end.setDate(end.getDate() + 1);
  const duration = Math.max(1, end.getTime() - start.getTime());
  return Math.max(0, Math.min(100, (timestamp - start.getTime()) / duration * 100));
}

export function BuildingInsights({
  buildingId,
  onClose,
}: {
  buildingId: string | null;
  onClose: () => void;
}) {
  const buildings = useWorldStore(selectBuildings);
  const residents = useWorldStore(selectResidents);
  const [now, setNow] = useState(() => Date.now());

  useEffect(() => {
    if (!buildingId) return;
    const interval = window.setInterval(() => setNow(Date.now()), 2_000);
    return () => window.clearInterval(interval);
  }, [buildingId]);

  useEffect(() => {
    if (!buildingId) return;
    const onKey = (event: KeyboardEvent) => {
      if (event.key === "Escape") onClose();
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [buildingId, onClose]);

  const building = buildings.find(item => item.id === buildingId);
  if (!building) return null;

  const progress = Math.round(buildingProgress(building, now) * 100);
  const floors = displayedFloors(building, now);
  const residentCount = residents.filter(item => item.homeId === building.id).length;
  const moss = Math.round(Math.max(0, Math.min(1, building.moss ?? 0)) * 100);
  const title = building.name || BUILDING_CATALOG.find(item => item.id === building.variantId)?.name || "Building";

  return (
    <aside className="eden-insights eden-building-insights" role="dialog" aria-label={`${title} building details`}>
      <div className="eden-insights-top">
        <span className="eden-eyebrow">EDEN / BUILDING</span>
        <button type="button" className="eden-insights-close" onClick={onClose} aria-label="Close building details">×</button>
      </div>
      <h2>{title}</h2>
      <div className="eden-grid-pill">{building.footprint.width} × {building.footprint.depth} <span>CELLS</span></div>
      <div className="eden-stat-grid">
        <div><strong>{floors}</strong><span>Floors</span></div>
        <div><strong>{moss}%</strong><span>Moss</span></div>
        <div><strong>{residentCount}</strong><span>Residents</span></div>
      </div>
      <div className="eden-insights-row eden-development-label">
        <span>Development</span><strong>{progress}%</strong>
      </div>
      <div className="eden-meter" role="progressbar" aria-label="Building development" aria-valuemin={0} aria-valuemax={100} aria-valuenow={progress}>
        <div className="eden-meter-fill" style={{ width: `${progress}%` }} />
      </div>
    </aside>
  );
}

// aero-insights-v3-pie: true proportional SVG pie slices.
const PIE_CENTER = 150;
const PIE_RADIUS = 132;

function piePoint(degrees: number, radius: number) {
  const radians = degrees * Math.PI / 180;
  return {
    x: PIE_CENTER + Math.cos(radians) * radius,
    y: PIE_CENTER + Math.sin(radians) * radius,
  };
}

function piePath(startDegrees: number, sweepDegrees: number) {
  const first = piePoint(startDegrees, PIE_RADIUS);
  const last = piePoint(startDegrees + sweepDegrees, PIE_RADIUS);
  const largeArc = sweepDegrees > 180 ? 1 : 0;
  return `M ${PIE_CENTER} ${PIE_CENTER} L ${first.x} ${first.y} A ${PIE_RADIUS} ${PIE_RADIUS} 0 ${largeArc} 1 ${last.x} ${last.y} Z`;
}

export function ActivityStats() {
  const activity = useWorldStore(selectActivity);
  const progression = useWorldStore(selectProgression);
  const [hovered, setHovered] = useState(false);
  const [focused, setFocused] = useState(false);
  const [pinned, setPinned] = useState(false);
  const [dismissed, setDismissed] = useState(false);
  const [now, setNow] = useState(() => Date.now());
  const open = !dismissed && (hovered || focused || pinned);

  useEffect(() => {
    const interval = window.setInterval(() => setNow(Date.now()), 30_000);
    return () => window.clearInterval(interval);
  }, []);

  // X and Escape suppress the hover/focus trigger until pointer exits,
  // focus leaves, or the status button is explicitly clicked again.
  const close = useCallback(() => {
    setDismissed(true);
    setPinned(false);
    setHovered(false);
    setFocused(false);
  }, []);

  useEffect(() => {
    if (!open) return;
    const onKey = (event: KeyboardEvent) => {
      if (event.key === "Escape") close();
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [open, close]);

  const rows = [
    { key: "productivity", label: "Productivity", points: activity.productivityPoints, milestones: progression.consumedMilestones.productivity },
    { key: "leisure", label: "Leisure", points: activity.leisurePoints, milestones: progression.consumedMilestones.leisure },
    { key: "idle", label: "Idle", points: activity.idlePoints, milestones: progression.consumedMilestones.idle },
  ];
  const total = rows.reduce((sum, row) => sum + row.points, 0);
  const dayProgress = cityDayPercentage(now);

  // Percentages and sector angles all come from the same real activity totals.
  let cursor = -90;
  const slices = rows.map(row => {
    const fraction = total > 0 ? Math.max(0, row.points) / total : 0;
    const sweep = Math.min(360, fraction * 360);
    const start = cursor;
    cursor += sweep;
    const mid = start + sweep / 2;
    const isSmall = fraction < 0.12;
    const position = piePoint(mid, isSmall ? 99 : fraction < 0.22 ? 88 : 76);
    return { ...row, fraction, sweep, start, isSmall, position };
  });

  return (
    <div
      className={`eden-activity-wrap${open ? " is-open" : ""}`}
      onMouseEnter={() => setHovered(true)}
      onMouseLeave={() => { setHovered(false); setDismissed(false); }}
      onFocus={() => setFocused(true)}
      onBlur={event => {
        if (!event.currentTarget.contains(event.relatedTarget as Node | null)) {
          setFocused(false);
          setDismissed(false);
        }
      }}
    >
      <button
        type="button"
        className="status"
        aria-expanded={open}
        aria-controls="eden-activity-stats"
        onClick={() => {
          if (open && pinned) close();
          else { setDismissed(false); setPinned(true); }
        }}
      >
        <i className={activity.category} />{activity.category}
      </button>
      <div
        id="eden-activity-stats"
        className="eden-insights eden-activity-popover"
        role="region"
        aria-label="Activity statistics"
        aria-hidden={!open}
      >
        <div className="eden-insights-top">
          <span className="eden-eyebrow">EDEN / ACTIVITY</span>
          <button type="button" tabIndex={open ? 0 : -1} className="eden-insights-close" onClick={close} aria-label="Close activity statistics">×</button>
        </div>
        <div className="eden-day-progress">
          <div className="eden-insights-row">
            <span>Day progress</span><strong>{dayProgress.toFixed(0)}%</strong>
          </div>
          <div className="eden-meter" role="progressbar" aria-label="Current city day progress" aria-valuemin={0} aria-valuemax={100} aria-valuenow={Math.round(dayProgress)}>
            <div className="eden-meter-fill eden-meter-day" style={{ width: `${dayProgress}%` }} />
          </div>
          <span className="eden-day-subtitle">4 AM → 4 AM</span>
        </div>
        <h2>Activity breakdown</h2>
        <div className="eden-pie-figure">
          <svg
            className="eden-pie-svg"
            viewBox="0 0 300 300"
            role="img"
            aria-label={slices.map(slice => `${slice.label}: ${Math.round(slice.fraction * 100)} percent, ${slice.milestones} milestones`).join('; ')}
          >
            <defs>
              <radialGradient id="eden-aero-pie-productivity" cx="24%" cy="16%" r="97%">
                <stop offset="0%" stopColor="#c1fff0" />
                <stop offset="45%" stopColor="#58dcbf" />
                <stop offset="100%" stopColor="#139baa" />
              </radialGradient>
              <radialGradient id="eden-aero-pie-leisure" cx="24%" cy="16%" r="97%">
                <stop offset="0%" stopColor="#fff8ce" />
                <stop offset="49%" stopColor="#f4c977" />
                <stop offset="100%" stopColor="#eaa65f" />
              </radialGradient>
              <radialGradient id="eden-aero-pie-idle" cx="24%" cy="16%" r="97%">
                <stop offset="0%" stopColor="#effbff" />
                <stop offset="45%" stopColor="#9ad5f0" />
                <stop offset="100%" stopColor="#62a4ca" />
              </radialGradient>
            </defs>
            {total <= 0 ? (
              <g>

                  ) : (

export function getCityDayKey(date = new Date()): string {
  const local = new Date(date);
  if (local.getHours() < 4) local.setDate(local.getDate() - 1);
  return `${local.getFullYear()}-${String(local.getMonth() + 1).padStart(2, "0")}-${String(local.getDate()).padStart(2, "0")}`;
}

export const DEFAULT_CITY_DAY_KEY = getCityDayKey();

export type DayPhase = "day" | "sunset" | "night";
export function getDayPhase(elapsedSeconds: number): DayPhase {
  const minute = ((elapsedSeconds % 600) + 600) % 600 / 60;
  if (minute < 5) return "day";
  if (minute < 6) return "sunset";
  return "night";
}

export function selectWeatherCondition(random: () => number = Math.random): Weather["condition"] {
  const roll = random();
  return roll < 0.62 ? "sunny" : roll < 0.88 ? "cloudy" : "rainy";
}
