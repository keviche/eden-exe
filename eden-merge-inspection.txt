// Matches the floor-count fallback used by buildBuilding in edenModels.ts.
function displayedFloors(building: Building, now: number): number {
  const config = BUILDING_CATALOG.find(item => item.id === building.variantId);
  const stage = growthStage(buildingProgress(building, now));
  const fallback = 1 + ((config?.floors ?? 1) - 1) * stage / 2;
  return Math.min(MAX_BUILDING_FLOORS, Math.max(1, Math.round(building.floors ?? fallback)));
}

  buildingId,
  onClose,

  buildingId: string | null;
  onClose: () => void;

  useEffect(() => {
    if (!buildingId) return;

  return (

      <div className="eden-insights-top">
        <span className="eden-eyebrow">BUILDING INSIGHTS</span>
        <button type="button" className="eden-insights-close" onClick={onClose} aria-label="Close building details">×</button>
      </div>
      <h2>{title}</h2>
      <p className="eden-muted">{building.footprint.width} × {building.footprint.depth} grid cells · {['Starting', 'Growing', 'Mature'][stage]} stage</p>
      <div className="eden-stat-grid">
        <div><strong>{floors}</strong><span>Floors</span></div>
        <div><strong>{moss}%</strong><span>Moss</span></div>
        <div><strong>{residentCount}</strong><span>Residents</span></div>
      </div>
      <div className="eden-insights-row"><span>Building development</span><strong>{progress}%</strong></div>
      <div className="eden-meter"><div className="eden-meter-fill" style={{ width: `${progress}%` }} /></div>
      <div className="eden-insights-row"><span>Share of built floor-space</span><strong>{share.toFixed(1)}%</strong></div>
      <div className="eden-meter"><div className="eden-meter-fill eden-meter-blue" style={{ width: `${share}%` }} /></div>
      <p className="eden-muted eden-insights-note">Built floor-space is estimated from grid footprint × floors, not physical square footage.</p>
      <h3>How this building grows</h3>
      <p className="eden-muted">Productivity milestones add floors or new buildings. Leisure can add vegetation and moss; activity points gradually grow moss on existing buildings.</p>
      <p className="eden-muted eden-insights-note">Buildings currently do not generate activity points. They reflect your city’s growth and provide homes for residents.</p>

// The city's statistics reset at local 4 AM; unlike a fixed 24h calculation,
// calculating consecutive local 4 AM boundaries also handles DST changes.
function cityDayPercentage(timestamp: number): number {
  const start = new Date(timestamp);
  start.setHours(4, 0, 0, 0);
  if (timestamp < start.getTime()) start.setDate(start.getDate() - 1);

  const end = new Date(start);
  end.setDate(end.getDate() + 1);
  const duration = Math.max(1, end.getTime() - start.getTime());
  return Math.max(0, Math.min(100, (timestamp - start.getTime()) / duration * 100));
}

// aero-insights-v3-pie: true proportional SVG pie slices.
const PIE_CENTER = 150;
const PIE_RADIUS = 132;

function piePoint(degrees: number, radius: number) {
  const radians = degrees * Math.PI / 180;
  return {
    x: PIE_CENTER + Math.cos(radians) * radius,
    y: PIE_CENTER + Math.sin(radians) * radius,
  };
}

function piePath(startDegrees: number, sweepDegrees: number) {
  const first = piePoint(startDegrees, PIE_RADIUS);
  const last = piePoint(startDegrees + sweepDegrees, PIE_RADIUS);
  const largeArc = sweepDegrees > 180 ? 1 : 0;
  return `M ${PIE_CENTER} ${PIE_CENTER} L ${first.x} ${first.y} A ${PIE_RADIUS} ${PIE_RADIUS} 0 ${largeArc} 1 ${last.x} ${last.y} Z`;
}

// Reflect a bubble off the inner glass boundaries.
// Reflection preserves movement rather than freezing at the wall.
function reflectWithinBounds(
  value: number,
  halfExtent: number,
  radius: number
): number {
  const limit = Math.max(0, halfExtent - radius - GLASS_CLEARANCE);
  if (limit === 0) return 0;

  const span = limit * 2;
  const period = span * 2;
  const wrapped = ((value + limit) % period + period) % period;

  return wrapped <= span
    ? -limit + wrapped
    : limit - (wrapped - span);
}

const SPARK_DIRECTIONS = Array.from({ length: SPARK_COUNT }, (_, index) => {

function BubbleMesh({ bubble }: { bubble: Bubble }) {
  const group = useRef<Group>(null);
  const shell = useRef<Group>(null);
  const film = useRef<Mesh>(null);
  const sparks = useRef<InstancedMesh>(null);
  const sparkMaterial = useRef<MeshBasicMaterial>(null);
  const clock = useThree(state => state.clock);
  const matrix = useMemo(() => new Matrix4(), []);
  const timeOffset = useRef(0);
  const popped = useRef<{ started: number; scale: number } | null>(null);

  function popBubble(event: ThreeEvent<MouseEvent>) {
    event.stopPropagation();
    const object = group.current;

    // Camera drags and right clicks should not pop bubbles.
    if (event.button !== 0 || event.delta > 5 || !object?.visible || popped.current) return;

    popped.current = {
      started: clock.getElapsedTime(),
      scale: object.scale.x,
    };

    if (shell.current) shell.current.visible = false;
  }

  useFrame(({ clock }) => {
    const object = group.current;
    if (!object) return;

    const t = clock.getElapsedTime();
    const pop = popped.current;

    if (pop) {
      const age = t - pop.started;

      if (age < RESPAWN_DELAY) {
        const progress = Math.min(1, age / POP_DURATION);
        object.visible = age < POP_DURATION;
        object.scale.setScalar(pop.scale);

        if (shell.current) shell.current.visible = false;

        if (sparks.current) {
          sparks.current.visible = age < POP_DURATION;

          const distance = bubble.radius * (0.65 + progress * 1.55);
          const size = bubble.radius * 0.095 * (1 - progress) + 0.002;

          SPARK_DIRECTIONS.forEach(([x, y, z], index) => {
            matrix.makeScale(size, size, size).setPosition(
              x * distance,
              y * distance,
              z * distance
            );
            sparks.current!.setMatrixAt(index, matrix);
          });

          sparks.current.instanceMatrix.needsUpdate = true;
        }

        if (sparkMaterial.current) {
          sparkMaterial.current.opacity = 0.9 * (1 - progress);
        }

        return;
      }

      // Restart at the bottom of the path with the existing fade-in.
      popped.current = null;
      timeOffset.current = t + bubble.phase - bubble.delay;
    }

    if (shell.current) shell.current.visible = true;
    if (sparks.current) sparks.current.visible = false;

    const elapsed = t + bubble.phase - bubble.delay - timeOffset.current;
    const cycle = ((elapsed % bubble.lifetime) + bubble.lifetime) % bubble.lifetime;

    // Each cycle begins near the bottom and travels up through the sky.
    const progress = cycle / bubble.lifetime;

    // Continuous X/Z drift plus overlapping waves creates a curved path.
    const x =
      bubble.x +
      bubble.driftX * cycle +
      Math.sin(cycle * 0.9 + bubble.phase) * bubble.sway +
      Math.sin(cycle * 0.37 + bubble.phase * 1.8) * bubble.sway * 0.45;

    const z =
      bubble.z +
      bubble.driftZ * cycle +
      Math.cos(cycle * 0.72 + bubble.phase) * bubble.sway * 0.65 +
      Math.sin(cycle * 0.43 + bubble.phase) * bubble.sway * 0.35;

    const y =
      MIN_HEIGHT +
      bubble.y +
      cycle * bubble.speed +
      Math.sin(cycle * 0.65 + bubble.phase) * 0.3;

    const safeX = reflectWithinBounds(x, HALF_TANK, bubble.radius);
    const safeZ = reflectWithinBounds(z, HALF_TANK, bubble.radius);

    object.position.set(safeX, y, safeZ);
    object.rotation.y = cycle * 0.16 + bubble.phase;
    object.rotation.z = Math.sin(cycle * 0.6 + bubble.phase) * 0.12;

    // Smoothly appear and disappear at the start/end of each cycle.
    const fadeIn = Math.min(1, progress / 0.12);
    const fadeOut = Math.min(1, (1 - progress) / 0.16);
    const opacity = Math.max(0, Math.min(fadeIn, fadeOut));

    object.scale.setScalar(Math.max(0.001, opacity));

    // Hide bubbles that have faded out.
    object.visible = opacity > 0.01;
  });

  const r = bubble.radius;

  return (
    <group ref={group} position={[bubble.x, MIN_HEIGHT + bubble.y, bubble.z]}>
      <group ref={shell}>
      {/* Iridescent, glossy soap-film shader */}
      <mesh
        ref={film}
        geometry={SPHERE}
        scale={r}
        onClick={popBubble}
        raycast={(raycaster, intersections) => {
          const object = group.current;

          // Ignore hidden, nearly vanished, or already popped bubbles.
          if (film.current && object?.visible && object.scale.x > 0.1 && !popped.current) {
            Mesh.prototype.raycast.call(film.current, raycaster, intersections);
          }
        }}
      >
        <BubbleFilm />
      </mesh>

      {/* Cyan reflective rim */}
      <mesh geometry={SPHERE} scale={r * 1.012} raycast={() => null}>
        <meshBasicMaterial
          color="#69dfff"
          transparent
          opacity={0.22}
          side={BackSide}
          depthWrite={false}
          blending={AdditiveBlending}
        />
      </mesh>

      {/* Iridescent pink sheen */}
      <mesh
        geometry={SPHERE}
        position={[r * 0.09, -r * 0.12, 0]}
        scale={[r * 0.94, r * 0.94, r * 0.94]}
        raycast={() => null}
      >
        <meshBasicMaterial
          color="#ffc0ed"
          transparent
          opacity={0.055}
          depthWrite={false}
        />
      </mesh>

      {/* Strong white glossy reflection */}
      <mesh
        position={[-r * 0.3, r * 0.43, r * 0.78]}
        rotation={[0, 0, -0.5]}
        scale={[r * 0.15, r * 0.33, r * 0.025]}
        raycast={() => null}

          color="#ffffff"

          opacity={0.85}
          depthWrite={false}

      {/* Secondary reflection */}

        rotation={[0, 0, 0.65]}
        scale={[r * 0.08, r * 0.17, r * 0.025]}

export default function SkyBubbles() {
  const bubbles = useMemo(() => {
    const random = randomGenerator(20261011);

    // Spread bubbles across a grid of regions, with random
    // positions inside each region to avoid visible patterns.
    const columns = 6;
    const rows = Math.ceil(COUNT / columns);

    const cells = Array.from(
      { length: columns * rows },
      (_, index) => index
    );

    // Shuffle the regions for a natural distribution.
    for (let i = cells.length - 1; i > 0; i--) {
      const j = Math.floor(random() * (i + 1));
      [cells[i], cells[j]] = [cells[j], cells[i]];
    }

    return Array.from({ length: COUNT }, (_, index): Bubble => {
      const cell = cells[index];
      const column = cell % columns;
      const row = Math.floor(cell / columns);

      const sizeRoll = random();

      const radius =
        sizeRoll > 0.87
          ? 0.9 + random() * 0.65
          : 0.18 + random() * 0.52;

      const speed = 0.55 + random() * 0.85;
      const lifetime = HEIGHT_RANGE / speed;

      // Full available footprint, accounting for bubble radius.
      const availableX = Math.max(
        0,
        HALF_TANK - radius - GLASS_CLEARANCE
      );
      const availableZ = Math.max(
        0,
        HALF_TANK - radius - GLASS_CLEARANCE
      );

      // Distribute each bubble in its own region.
      const x =
        -availableX +
        ((column + 0.15 + random() * 0.7) / columns) *
          availableX * 2;

      const z =
        -availableZ +
        ((row + 0.15 + random() * 0.7) / rows) *
          availableZ * 2;

      return {
        x,
        y: random() * HEIGHT_RANGE,
        z,
        radius,
        speed,
        driftX: (random() - 0.5) * 0.75,
        driftZ: (random() - 0.5) * 0.65,
        sway: 0.35 + random() * 0.8,
        phase: random() * lifetime,
        lifetime,
        delay: random() * lifetime,
      };
    });
  }, []);

  return (
    <group>
      {bubbles.map((bubble, index) => (
        <BubbleMesh key={index} bubble={bubble} />
      ))}
    </group>
  );
}

function applyMilestones(worldInput: WorldState, at: number): WorldState {
  let world = worldInput;
  const consumedMilestones = { ...world.progression.consumedMilestones };
  for (const category of ["productivity", "leisure", "idle"] as const) {
    const total = world.activity[ACTIVITY_KEY[category]];
    const earned = Math.floor(total / ACTIVITY_MILESTONE_POINTS);
    let consumed = consumedMilestones[category];
    while (consumed < earned) {
      let changed = false;
      if (category === "productivity") {
        const eligible = world.buildings.find(building => (building.floors ?? 1) < MAX_BUILDING_FLOORS);
        if (eligible) {
          world = { ...world, buildings: world.buildings.map(building => building.id === eligible.id ? { ...building, floors: Math.min(MAX_BUILDING_FLOORS, (building.floors ?? 1) + 1), growth: 1 } : building) };
          changed = true;
        } else {
          const building = createBuilding(world, at);
          if (building) { world = { ...world, buildings: [...world.buildings, building] }; changed = true; }
        }
      } else if (category === "leisure") {
        const buildingCount = world.buildings.length;
        const treeCount = world.plants.filter(
          plant => plant.species === "tree"
        ).length;

        // Roughly 1 tree per building, with some variation.
        const maxTrees = Math.max(
          1,
          Math.ceil(buildingCount * 1.2)
        );

        // Limit total nature so buildings remain visually prominent.
        const maxNature = Math.max(
          2,
          Math.ceil(buildingCount * 1.8)
        );

        if (world.plants.length < maxNature) {
          let plant = createPlant(world, at);

          if (plant) {
            if (
              plant.species === "tree" &&
              treeCount >= maxTrees
            ) {
              const alternatives = [
                "shrub",
                "fern",
                "flower",
                "grass",
              ] as const;

              const species =
                alternatives[
                  Math.floor(Math.random() * alternatives.length)
                ]!;

              plant = {
                ...plant,
                species,
                variantId: species,
              };
            }

            world = {
              ...world,
              plants: [...world.plants, plant],
            };

            changed = true;
          }
        }

        // Existing buildings continue developing moss and vines.
        if (
          earned >= 3 &&
          world.buildings.some(
            building => (building.moss ?? 0) < 1
          )
        ) {
          world = {
            ...world,
            buildings: world.buildings.map(building => ({
              ...building,
              moss: Math.min(
                1,
                Math.max(
                  building.moss ?? 0,
                  (earned - 2) / 4
                )
              ),
            })),
          };

          changed = true;
        }

      } else if (world.residents.length < MAX_RESIDENTS) {
        const cell = findOpenResidentCell(world.buildings, world.plants, world.residents.map(resident => resident.position));
        if (!cell) break;
        const building = world.buildings[Math.floor(Math.random() * Math.max(1, world.buildings.length))];
        const resident: Resident = { type: "resident", id: createId("resident"), name: `Resident ${world.residents.length + 1}`, position: residentWorldPosition(cell), activity: "walking", homeId: building?.id, route: [], createdAt: at };
        world = { ...world, residents: [...world.residents, resident] }; changed = true;
      }
      if (!changed) break;
      consumed++;
    }
    consumedMilestones[category] = consumed;
  }
  return { ...world, progression: { ...world.progression, consumedMilestones } };
}

function roundedPerimeter(halfWidth: number, halfDepth: number, radius: number, y: number) {
  const points: Vector3[] = [];
  const corners = [
    { x: halfWidth - radius, z: -halfDepth + radius, from: -90 },
    { x: halfWidth - radius, z: halfDepth - radius, from: 0 },
    { x: -halfWidth + radius, z: halfDepth - radius, from: 90 },
    { x: -halfWidth + radius, z: -halfDepth + radius, from: 180 },
  ];
  for (const corner of corners) for (let step = 0; step <= 10; step++) {
    const angle = (corner.from + step * 9) * Math.PI / 180;
    points.push(new Vector3(corner.x + Math.cos(angle) * radius, y, corner.z + Math.sin(angle) * radius));
  }
  return points;
}

function tankInsetAtHeight(height: number) {
  const h = Math.max(
    0,
    Math.min(BASE_CURVE_RADIUS, height)
  );
  const dy = BASE_CURVE_RADIUS - h;

  return BASE_CURVE_RADIUS - Math.sqrt(
    Math.max(0, BASE_CURVE_RADIUS ** 2 - dy ** 2)
  );
}

function createTankWallGeometry() {
  const positions: number[] = [];
  const indices: number[] = [];
  const contours: Vector3[][] = [];
  const bottomY = -TERRAIN_LIFT;

  // Generate a quarter-circle transition between the
  // horizontal glass floor and the vertical glass walls.
  for (let step = 0; step <= 16; step++) {
    const angle = step / 16 * Math.PI / 2;

    const height =
      BASE_CURVE_RADIUS * (1 - Math.cos(angle));

    const inset =
      BASE_CURVE_RADIUS * (1 - Math.sin(angle));

    contours.push(
      roundedPerimeter(
        TANK_WIDTH / 2 - inset,
        TANK_WIDTH / 2 - inset,
        TANK_RADIUS - inset,
        bottomY + height
      )
    );
  }

  // Continue vertically to the open top.
  contours.push(
    roundedPerimeter(
      TANK_WIDTH / 2,
      TANK_WIDTH / 2,
      TANK_RADIUS,
      TANK_HEIGHT
    )
  );

  const count = contours[0]!.length;

  for (const outline of contours) {
    if (outline.length !== count) {
      throw new Error("Inconsistent tank outline");
    }

    for (const point of outline) {
      positions.push(point.x, point.y, point.z);
    }
  }

  // Join neighboring contours into a continuous surface.
  for (let level = 0; level < contours.length - 1; level++) {
    for (let i = 0; i < count; i++) {
      const next = (i + 1) % count;

      const a = level * count + i;
      const b = level * count + next;
      const c = (level + 1) * count + i;
      const d = (level + 1) * count + next;

      indices.push(a, c, d, a, d, b);
    }
  }

  // Close the underside with transparent glass.
  // This also prevents an open hole when viewed from below.
  const center = positions.length / 3;
  positions.push(0, bottomY, 0);

  for (let i = 0; i < count; i++) {
    indices.push(center, i, (i + 1) % count);
  }

  const geometry = new BufferGeometry();

  geometry.setAttribute(
    "position",
    new Float32BufferAttribute(positions, 3)
  );

  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  geometry.computeBoundingSphere();

  return geometry;
}

function WaterCourse({ seed }: { seed: number }) {
  const layout = useMemo(() => {
    const random = waterRandom(seed);
    const range = (min: number, max: number) =>
      min + random() * (max - min);

    const edge = BOARD_SIZE * 0.46;
    const riverPositions: number[] = [];

    // Randomized winding river.
    const startX = range(-BOARD_SIZE * 0.17, BOARD_SIZE * 0.17);
    const endX = range(-BOARD_SIZE * 0.17, BOARD_SIZE * 0.17);
    const bend1 = range(-BOARD_SIZE * 0.13, BOARD_SIZE * 0.13);
    const bend2 = range(-BOARD_SIZE * 0.07, BOARD_SIZE * 0.07);
    const phase = range(0, Math.PI * 2);
    const width = range(0.42, 0.78);

    const riverPoint = (t: number) => {
      const z = -edge + t * edge * 2;

      const rawX =
        startX * (1 - t) +
        endX * t +
        Math.sin(t * Math.PI * 2 + phase) * bend1 +
        Math.sin(t * Math.PI * 4 + phase * 0.6) * bend2;

      const w = width *
        (1 + 0.18 * Math.sin(t * Math.PI * 5 + phase));

      const x = Math.max(
        -edge + w,
        Math.min(edge - w, rawX)
      );

      return { x, z, w };
    };

    const steps = 100;

    for (let i = 0; i < steps; i++) {
      const a = riverPoint(i / steps);
      const b = riverPoint((i + 1) / steps);

      riverPositions.push(
        a.x - a.w, 0.065, a.z,
        b.x - b.w, 0.065, b.z,
        a.x + a.w, 0.065, a.z,

        a.x + a.w, 0.065, a.z,
        b.x - b.w, 0.065, b.z,
        b.x + b.w, 0.065, b.z
      );
    }

    const river = new BufferGeometry();
    river.setAttribute(
      "position",
      new Float32BufferAttribute(riverPositions, 3)
    );
    river.computeVertexNormals();

    // Organic shoreline geometry for lakes and ponds.
    const makeWaterBody = (
      x: number,
      z: number,
      radiusX: number,
      radiusZ: number
    ) => {
      const positions: number[] = [];
      const segments = 72;
      const phase1 = range(0, Math.PI * 2);
      const phase2 = range(0, Math.PI * 2);
      const wobble = range(0.06, 0.13);

      const point = (angle: number) => {
        const variation =
          1 +
          wobble * Math.sin(angle * 3 + phase1) +
          wobble * 0.45 * Math.cos(angle * 5 + phase2);

        return [
          x + Math.cos(angle) * radiusX * variation,
          0.078,
          z + Math.sin(angle) * radiusZ * variation
        ];
      };

      for (let i = 0; i < segments; i++) {
        const a = i / segments * Math.PI * 2;
        const b = (i + 1) / segments * Math.PI * 2;

        positions.push(
          x, 0.078, z,
          ...point(b),
          ...point(a)
        );
      }

      const geometry = new BufferGeometry();
      geometry.setAttribute(
        "position",
        new Float32BufferAttribute(positions, 3)
      );
      geometry.computeVertexNormals();

      return geometry;
    };

    // Put the lake and pond on opposite sides of the river.
    const lakeSide = random() < 0.5 ? -1 : 1;

    const lakeRadiusX = range(1.8, 3.1);
    const lakeRadiusZ = range(1.4, 2.5);

    const lakeX = lakeSide *
      range(BOARD_SIZE * 0.20, BOARD_SIZE * 0.29);
    const lakeZ = range(
      -BOARD_SIZE * 0.25,
      BOARD_SIZE * 0.25
    );

    const pondRadiusX = range(2.0, 3.8);
    const pondRadiusZ = range(1.7, 3.2);

    const pondX = -lakeSide *
      range(BOARD_SIZE * 0.20, BOARD_SIZE * 0.30);
    const pondZ = range(
      -BOARD_SIZE * 0.30,
      BOARD_SIZE * 0.30
    );


    const pond = makeWaterBody(
      pondX, pondZ, pondRadiusX, pondRadiusZ
    );

    return { river, pond };
  }, [seed]);

  useEffect(() => {
    return () => {
      layout.river.dispose();
      
      layout.pond.dispose();
    };
  }, [layout]);

  return (
    <group>
      <mesh geometry={layout.river} receiveShadow>
        <WaterSurface kind="river" />
      </mesh>

      

      <mesh geometry={layout.pond} receiveShadow>
        <WaterSurface kind="pond" />
      </mesh>
    </group>
  );
}

function reflectWithinBounds(
  value: number,
  halfExtent: number,
  radius: number
): number {
  const limit = Math.max(0, halfExtent - radius - GLASS_CLEARANCE);
  if (limit === 0) return 0;

  const span = limit * 2;
  const period = span * 2;
  const wrapped = ((value + limit) % period + period) % period;

  return wrapped <= span
    ? -limit + wrapped

}

      // dimensions so a rectangular building still fits its recorded plot.
      const { width, depth } = buildingDimensions(item.footprint, item.rotation);
      const progress = buildingProgress(item, now);
      const stage = growthStage(progress);
      const model = buildingModel(item.variantId, width, depth, stage, hashId(item.id), item.floors, item.moss);
      // Horizontal size stays fixed so growth cannot consume neighboring plots.
      const buildingVariation = 0.75 + (hashId(item.id) % 100) / 100 * 0.8;
    const root = compose(
      [at.x, 0, at.z],
      [0, item.rotation, 0],
      [1, (0.65 + progress * 0.35) * buildingVariation, 1]
    );
      placeModel(model, root, parts);

      // Climbing vines and leaves appear as moss increases.
      const moss = Math.max(0, Math.min(1, item.moss ?? 0));

      if (moss > 0.05) {
        const height = (item.floors ?? 1) * 0.55 + 0.5;
        const vineCount = Math.ceil(moss * 12);

        for (let i = 0; i < vineCount; i++) {
          const side = i % 4;
          const fraction = ((i * 7) % 11) / 11 - 0.5;

          const localX =
            side < 2
              ? fraction * width * 0.85
              : (side === 2 ? -1 : 1) * width * 0.46;

          const localZ =
            side < 2
              ? (side === 0 ? -1 : 1) * depth * 0.46
              : fraction * depth * 0.85;

          const angle = item.rotation;

          const x =
            at.x +
            localX * Math.cos(angle) +
            localZ * Math.sin(angle);

          const z =
            at.z -
            localX * Math.sin(angle) +
            localZ * Math.cos(angle);

          const vineHeight =
            height * moss * (0.5 + (i % 4) * 0.12);

          parts.push({
            shape: "stem",
            finish: "matte",
            color: i % 3 === 0 ? "#397b3c" : "#4d9b4a",
            matrix: compose(
              [x, vineHeight / 2, z],
              [0, 0, 0],
              [
                0.035 + moss * 0.035,
                vineHeight,
                0.035 + moss * 0.035,
              ]
            ),
          });

          const leafCount = Math.ceil(moss * 6);

          for (let j = 0; j < leafCount; j++) {
            const leafY =
              ((j + 1) / (leafCount + 1)) * vineHeight;

            parts.push({
              shape: "leaf",
              finish: "matte",
              color: j % 2 === 0 ? "#56a94c" : "#7bc768",
              matrix: compose(
                [
                  x + (j % 2 === 0 ? -0.06 : 0.06),
                  leafY,
                  z,
                ],
                [0, 0, j % 2 === 0 ? -0.5 : 0.5],
                [
                  0.09 + moss * 0.06,
                  0.045 + moss * 0.035,
                  0.065,
                ]
              ),
            });
          }
        }
      }

    }
    return batchParts(parts);
  }, [buildings, now]);

  return <BatchLayer batches={batches} night={night} />;
}

function PlantsLayer({ night }: { night: boolean }) {
  const plants = useWorldStore(selectPlants);
  const batches = useMemo(() => {
    const parts: Part[] = [];
    for (const item of plants) {
      const at = cellPoint(item.gridPosition.x, item.gridPosition.z);
      const seed = hashId(item.id);
      const speciesScale = item.species === "tree" ? 2.2 : item.species === "shrub" ? 1.6 : 1.35;
      const growth = Math.max(0, Math.min(3, item.growth));

    const natureMultiplier =
      item.species === "tree" ? 1.65 :
      item.species === "shrub" ? 1.4 :
      item.species === "fern" ? 1.35 :
      item.species === "flower" ? 1.5 :
      1.25;

    const individualSize = 0.7 + (seed % 100) / 100 * 0.9;

    const scale =
      (0.25 + growth * 0.65) *
      individualSize *
      (CELL_SIZE / 1.5) *
      speciesScale *
      natureMultiplier;
      const root = compose([at.x, 0, at.z], [0, seed % 360 * Math.PI / 180, 0], [scale, scale, scale]);
      placeModel(plantModel(item.species, seed), root, parts);
    }
    return batchParts(parts);
  }, [plants]);
  return <BatchLayer batches={batches} night={night} />;
}

function createGableGeometry() {
  // A low, elongated roof with a continuous ridge, matching the cottage forms

function roundedPerimeter(halfWidth: number, halfDepth: number, radius: number, y: number) {
  const points: Vector3[] = [];
  const corners = [
    { x: halfWidth - radius, z: -halfDepth + radius, from: -90 },
    { x: halfWidth - radius, z: halfDepth - radius, from: 0 },
    { x: -halfWidth + radius, z: halfDepth - radius, from: 90 },
    { x: -halfWidth + radius, z: -halfDepth + radius, from: 180 },
  ];
  for (const corner of corners) for (let step = 0; step <= 10; step++) {
    const angle = (corner.from + step * 9) * Math.PI / 180;
    points.push(new Vector3(corner.x + Math.cos(angle) * radius, y, corner.z + Math.sin(angle) * radius));
  }
  return points;
}

// Matching rounded corners for soil and glass.
// The glass bends smoothly underneath the terrarium.
const BASE_CURVE_RADIUS = 0.7;
const SOIL_CLEARANCE = 0.32;

function tankInsetAtHeight(height: number) {
  const h = Math.max(
    0,
    Math.min(BASE_CURVE_RADIUS, height)
  );
  const dy = BASE_CURVE_RADIUS - h;

  return BASE_CURVE_RADIUS - Math.sqrt(
    Math.max(0, BASE_CURVE_RADIUS ** 2 - dy ** 2)
  );
}

function createTankWallGeometry() {
  const positions: number[] = [];
  const indices: number[] = [];
  const contours: Vector3[][] = [];
  const bottomY = -TERRAIN_LIFT;

  // Generate a quarter-circle transition between the
  // horizontal glass floor and the vertical glass walls.
  for (let step = 0; step <= 16; step++) {
    const angle = step / 16 * Math.PI / 2;

    const height =
      BASE_CURVE_RADIUS * (1 - Math.cos(angle));

    const inset =
      BASE_CURVE_RADIUS * (1 - Math.sin(angle));

    contours.push(
      roundedPerimeter(
        TANK_WIDTH / 2 - inset,
        TANK_WIDTH / 2 - inset,
        TANK_RADIUS - inset,
        bottomY + height
      )
    );
  }

  // Continue vertically to the open top.
  contours.push(
    roundedPerimeter(
      TANK_WIDTH / 2,
      TANK_WIDTH / 2,
      TANK_RADIUS,
      TANK_HEIGHT
    )
  );

  const count = contours[0]!.length;

  for (const outline of contours) {
    if (outline.length !== count) {
      throw new Error("Inconsistent tank outline");
    }

    for (const point of outline) {
      positions.push(point.x, point.y, point.z);
    }
  }

  // Join neighboring contours into a continuous surface.
  for (let level = 0; level < contours.length - 1; level++) {
    for (let i = 0; i < count; i++) {
      const next = (i + 1) % count;

      const a = level * count + i;
      const b = level * count + next;
      const c = (level + 1) * count + i;
      const d = (level + 1) * count + next;

      indices.push(a, c, d, a, d, b);
    }
  }

  // Close the underside with transparent glass.
  // This also prevents an open hole when viewed from below.
  const center = positions.length / 3;
  positions.push(0, bottomY, 0);

  for (let i = 0; i < count; i++) {
    indices.push(center, i, (i + 1) % count);
  }

  const geometry = new BufferGeometry();

  geometry.setAttribute(
    "position",
    new Float32BufferAttribute(positions, 3)
  );

  geometry.setIndex(indices);
  geometry.computeVertexNormals();
  geometry.computeBoundingSphere();

  return geometry;
}

function createRoundedGroundShape() {
  // The soil shares the exact corner centers of the glass.
  // A small inset ensures the dirt never protrudes outside.
  const half = TANK_WIDTH / 2 - SOIL_CLEARANCE;
  const radius = TANK_RADIUS - SOIL_CLEARANCE;

  const shape = new ThreeShape();

  shape.moveTo(-half + radius, -half);

  shape.lineTo(half - radius, -half);
  shape.quadraticCurveTo(
    half, -half, half, -half + radius
  );

  shape.lineTo(half, half - radius);
  shape.quadraticCurveTo(
    half, half, half - radius, half
  );

  shape.lineTo(-half + radius, half);
  shape.quadraticCurveTo(
    -half, half, -half, half - radius
  );

  shape.lineTo(-half, -half + radius);
  shape.quadraticCurveTo(
    -half, -half, -half + radius, -half
  );

  return shape;
}

  }

function waterRandom(seed: number) {
  let state = seed >>> 0;

  return () => {
    state = (Math.imul(state, 1664525) + 1013904223) >>> 0;
    return state / 4294967296;
  };
}

const WaterCourse = memo(function WaterCourse({ seed }: { seed: number }) {
  const layout = useMemo(() => {
    const random = waterRandom(seed);
    const range = (min: number, max: number) =>
      min + random() * (max - min);

    const edge = BOARD_SIZE * 0.46;
    const riverPositions: number[] = [];

    // Randomized winding river.
    const startX = range(-BOARD_SIZE * 0.17, BOARD_SIZE * 0.17);
    const endX = range(-BOARD_SIZE * 0.17, BOARD_SIZE * 0.17);
    const bend1 = range(-BOARD_SIZE * 0.13, BOARD_SIZE * 0.13);
    const bend2 = range(-BOARD_SIZE * 0.07, BOARD_SIZE * 0.07);
    const phase = range(0, Math.PI * 2);
    const width = range(0.42, 0.78);

    const riverPoint = (t: number) => {
      const z = -edge + t * edge * 2;

      const rawX =
        startX * (1 - t) +
        endX * t +
        Math.sin(t * Math.PI * 2 + phase) * bend1 +
        Math.sin(t * Math.PI * 4 + phase * 0.6) * bend2;

      const w = width *
        (1 + 0.18 * Math.sin(t * Math.PI * 5 + phase));

      const x = Math.max(
        -edge + w,
        Math.min(edge - w, rawX)
      );

      return { x, z, w };
    };

    const steps = 100;

    for (let i = 0; i < steps; i++) {
      const a = riverPoint(i / steps);
      const b = riverPoint((i + 1) / steps);

      riverPositions.push(
        a.x - a.w, 0.065, a.z,
        b.x - b.w, 0.065, b.z,
        a.x + a.w, 0.065, a.z,

        a.x + a.w, 0.065, a.z,
        b.x - b.w, 0.065, b.z,
        b.x + b.w, 0.065, b.z
      );
    }

    const river = new BufferGeometry();
    river.setAttribute(
      "position",
      new Float32BufferAttribute(riverPositions, 3)
    );
    river.computeVertexNormals();

    // Organic shoreline geometry for lakes and ponds.
    const makeWaterBody = (
      x: number,
      z: number,
      radiusX: number,
      radiusZ: number
    ) => {
      const positions: number[] = [];
      const segments = 72;
      const phase1 = range(0, Math.PI * 2);
      const phase2 = range(0, Math.PI * 2);
      const wobble = range(0.06, 0.13);

      const point = (angle: number) => {
        const variation =
          1 +
          wobble * Math.sin(angle * 3 + phase1) +
          wobble * 0.45 * Math.cos(angle * 5 + phase2);

        return [
          x + Math.cos(angle) * radiusX * variation,
          0.078,
          z + Math.sin(angle) * radiusZ * variation
        ];
      };

      for (let i = 0; i < segments; i++) {
        const a = i / segments * Math.PI * 2;
        const b = (i + 1) / segments * Math.PI * 2;

        positions.push(
          x, 0.078, z,
          ...point(b),
          ...point(a)
        );
      }

      const geometry = new BufferGeometry();
      geometry.setAttribute(
        "position",
        new Float32BufferAttribute(positions, 3)
      );
      geometry.computeVertexNormals();

      return geometry;
    };

    // Put the lake and pond on opposite sides of the river.
    const lakeSide = random() < 0.5 ? -1 : 1;

    const lakeRadiusX = range(1.8, 3.1);
    const lakeRadiusZ = range(1.4, 2.5);

    const lakeX = lakeSide *
      range(BOARD_SIZE * 0.20, BOARD_SIZE * 0.29);
    const lakeZ = range(
      -BOARD_SIZE * 0.25,
      BOARD_SIZE * 0.25
    );

    const pondRadiusX = range(2.0, 3.8);
    const pondRadiusZ = range(1.7, 3.2);

    const pondX = -lakeSide *
      range(BOARD_SIZE * 0.20, BOARD_SIZE * 0.30);
    const pondZ = range(
      -BOARD_SIZE * 0.30,
      BOARD_SIZE * 0.30
    );


    const pond = makeWaterBody(
      pondX, pondZ, pondRadiusX, pondRadiusZ
    );

    return { river, pond };
  }, [seed]);

  useEffect(() => {
    return () => {
      layout.river.dispose();
      
      layout.pond.dispose();
    };
  }, [layout]);

  return (
    <group>
      <mesh geometry={layout.river} receiveShadow>
        <WaterSurface kind="river" />
      </mesh>

      

      <mesh geometry={layout.pond} receiveShadow>
        <WaterSurface kind="pond" />
      </mesh>
    </group>
  );
});

const MossBanks = memo(function MossBanks({ night }: { night: boolean }) {
  const ref = useRef<InstancedMesh>(null);
  const invalidate = useThree(state => state.invalidate);
  const geometry = useMemo(() => new SphereGeometry(1, 12, 8), []);
  const patches = useMemo(() => Array.from({ length: 176 }, (_, index) => {
    const angle = index / 136 * Math.PI * 2;
    const cosine = Math.cos(angle), sine = Math.sin(angle);
    const edge = (BOARD_SIZE * (0.41 + ((index * 17) % 10) / 100)) / Math.max(Math.abs(cosine), Math.abs(sine));
    if (index >= 136) {
      const waterAngle = (index - 136) / 40 * Math.PI * 2;
      return { x: 6 + Math.cos(waterAngle) * (3.7 + (index % 3) * 0.16), z: BOARD_SIZE * 0.29 + Math.sin(waterAngle) * (2.15 + (index % 4) * 0.13), sx: 0.62 + index % 3 * 0.12, sz: 0.57 + index % 4 * 0.1, color: index % 2 };
    }
    return { x: cosine * edge, z: sine * edge, sx: 0.74 + (index * 7 % 9) * 0.105, sz: 0.72 + (index * 5 % 8) * 0.11, color: index % 4 };
  }), []);
  useLayoutEffect(() => {
    const mesh = ref.current;
    if (!mesh) return;
    const matrix = new Matrix4();
    const colors = ["#47b85b", "#55c564", "#68ce69", "#3da857"];
    patches.forEach((patch, index) => {
      matrix.makeScale(patch.sx, 0.13 + index % 3 * 0.018, patch.sz).setPosition(patch.x, 0.075, patch.z);
      mesh.setMatrixAt(index, matrix);
      mesh.setColorAt(index, new Color(colors[patch.color]!));
    });
    mesh.count = patches.length;
    mesh.instanceMatrix.needsUpdate = true;
    if (mesh.instanceColor) mesh.instanceColor.needsUpdate = true;
    mesh.computeBoundingSphere();
    invalidate();
  }, [patches, invalidate]);
  return <instancedMesh ref={ref} args={[geometry, undefined, patches.length]} receiveShadow>
    <meshStandardMaterial color={night ? "#c0e4c9" : "#ffffff"} roughness={0.88} />
  </instancedMesh>;
});

/** A single draw batch for one geometry/material combination. */
const InstanceBatch = memo(function InstanceBatch({
  batch, night, owners, onSelectBuilding,
}: {
  batch: Batch;
  night: boolean;
  owners?: string[];
  onSelectBuilding?: (id: string) => void;
}) {
  const meshRef = useRef<InstancedMesh>(null);
  const invalidate = useThree(state => state.invalidate);
  // Spare capacity limits GPU buffer recreation as new objects are added.
  const capacity = Math.max(16, 2 ** Math.ceil(Math.log2(Math.max(1, batch.parts.length))));

  useLayoutEffect(() => {
    const mesh = meshRef.current;
    if (!mesh) return;
    const matrix = new Matrix4();
    const color = new Color();
    mesh.count = batch.parts.length;
    batch.parts.forEach((part, index) => {
      mesh.setMatrixAt(index, matrix.fromArray(part.matrix));
      mesh.setColorAt(index, color.set(part.color));
    });
    mesh.instanceMatrix.needsUpdate = true;
    if (mesh.instanceColor) mesh.instanceColor.needsUpdate = true;
    // Includes distant plots after additions, movement, or growth.
    mesh.computeBoundingSphere();
    mesh.computeBoundingBox();
    invalidate();
  }, [batch.parts, capacity, invalidate]);

  const glass = batch.finish === "glass";
  const greenhouse = batch.finish === "greenhouse";
  const solar = batch.finish === "solar";
  const metal = batch.finish === "metal";

  return (
    <instancedMesh
      ref={meshRef}
      args={[GEOMETRIES[batch.shape], undefined, capacity]}
      onClick={owners && onSelectBuilding ? event => {
        if (event.button !== 0 || event.delta > 5) return;
        const id = owners[event.instanceId ?? -1];
        if (!id) return;
        event.stopPropagation();
        onSelectBuilding(id);
      } : undefined}
      onPointerOver={owners ? () => { document.body.style.cursor = 'pointer'; } : undefined}
      onPointerOut={owners ? () => { document.body.style.cursor = ''; } : undefined}
      castShadow
      receiveShadow
    >
      <meshPhysicalMaterial
        color="#ffffff"
        transparent={glass || greenhouse}
        opacity={glass ? 0.4 : greenhouse ? 0.18 : 1}
        depthWrite={!glass && !greenhouse}
        side={2}
        roughness={glass || greenhouse ? 0.12 : solar ? 0.2 : metal ? 0.32 : 0.78}
        metalness={glass || greenhouse ? 0.04 : solar ? 0.3 : metal ? 0.36 : 0}
        emissive={glass ? "#fff0ab" : greenhouse ? "#8af5df" : "#000000"}
        emissiveIntensity={glass ? (night ? 0.58 : 0.16) : greenhouse ? (night ? 0.12 : 0.045) : 0}
        clearcoat={glass || greenhouse ? 0.7 : solar ? 0.9 : 0.12}
        clearcoatRoughness={glass || greenhouse ? 0.06 : solar ? 0.12 : 0.32}
        flatShading={!glass && !greenhouse && !metal}
      />
    </instancedMesh>
  );
});

function BatchLayer({ batches, night }: { batches: Batch[]; night: boolean }) {
  return <>{batches.map(batch => <InstanceBatch key={batch.key} batch={batch} night={night} />)}</>;
}

/** Store growth overrides age-based growth if the simulation supplies it. */
function BuildingsLayer({ night, onSelectBuilding }: { night: boolean; onSelectBuilding: (id: string) => void }) {
  const buildings = useWorldStore(selectBuildings);
  const [now, setNow] = useState(() => Date.now());

  useEffect(() => {
    const growing = buildings.filter(item => !hasExplicitGrowth(item) && buildingProgress(item, Date.now()) < 1);
    if (growing.length === 0) return;
    const timer = window.setInterval(() => {
      const next = Date.now();
      setNow(next);
      if (growing.every(item => buildingProgress(item, next) >= 1)) window.clearInterval(timer);
    }, GROWTH_REFRESH_MS);
    return () => window.clearInterval(timer);
  }, [buildings]);

  const { batches, ownerByBatch } = useMemo(() => {
    const parts: Part[] = [];
    const ownerByPart: string[] = [];
    for (const item of buildings) {
      const firstPart = parts.length;
      const at = cellPoint(
        item.gridPosition.x + (item.footprint.width - 1) / 2,
        item.gridPosition.z + (item.footprint.depth - 1) / 2,
      );
      // Footprints are occupied grid dimensions. A quarter-turn swaps the local
      // dimensions so a rectangular building still fits its recorded plot.
      const { width, depth } = buildingDimensions(item.footprint, item.rotation);
      const progress = buildingProgress(item, now);
      const stage = growthStage(progress);
      const model = buildingModel(item.variantId, width, depth, stage, hashId(item.id), item.floors, item.moss);
      // Horizontal size stays fixed so growth cannot consume neighboring plots.
      const buildingVariation = 0.75 + (hashId(item.id) % 100) / 100 * 0.8;
    const root = compose(
      [at.x, 0, at.z],
      [0, item.rotation, 0],
      [1, (0.65 + progress * 0.35) * buildingVariation, 1]
    );
      placeModel(model, root, parts);

      // Climbing vines and leaves appear as moss increases.
      const moss = Math.max(0, Math.min(1, item.moss ?? 0));

      if (moss > 0.05) {
        const height = (item.floors ?? 1) * 0.55 + 0.5;
        const vineCount = Math.ceil(moss * 12);

        for (let i = 0; i < vineCount; i++) {
          const side = i % 4;
          const fraction = ((i * 7) % 11) / 11 - 0.5;

          const localX =
            side < 2
              ? fraction * width * 0.85
              : (side === 2 ? -1 : 1) * width * 0.46;

          const localZ =
            side < 2
              ? (side === 0 ? -1 : 1) * depth * 0.46
              : fraction * depth * 0.85;

          const angle = item.rotation;

          const x =
            at.x +
            localX * Math.cos(angle) +
            localZ * Math.sin(angle);

          const z =
            at.z -
            localX * Math.sin(angle) +
            localZ * Math.cos(angle);

          const vineHeight =
            height * moss * (0.5 + (i % 4) * 0.12);

          parts.push({
            shape: "stem",
            finish: "matte",
            color: i % 3 === 0 ? "#397b3c" : "#4d9b4a",
            matrix: compose(
              [x, vineHeight / 2, z],
              [0, 0, 0],
              [
                0.035 + moss * 0.035,
                vineHeight,
                0.035 + moss * 0.035,
              ]
            ),
          });

          const leafCount = Math.ceil(moss * 6);

          for (let j = 0; j < leafCount; j++) {
            const leafY =
              ((j + 1) / (leafCount + 1)) * vineHeight;

            parts.push({
              shape: "leaf",
              finish: "matte",
              color: j % 2 === 0 ? "#56a94c" : "#7bc768",
              matrix: compose(
                [
                  x + (j % 2 === 0 ? -0.06 : 0.06),
                  leafY,
                  z,
                ],
                [0, 0, j % 2 === 0 ? -0.5 : 0.5],
                [
                  0.09 + moss * 0.06,
                  0.045 + moss * 0.035,
                  0.065,
                ]
              ),
            });
          }
        }
      }

      // Parts appended by this building (including moss/vines) share its ID.
      for (let i = firstPart; i < parts.length; i++) ownerByPart.push(item.id);
    }
    const ownerByBatch = new Map<string, string[]>();
    parts.forEach((part, index) => {
      const key = `${part.shape}:${part.finish}`;
      let list = ownerByBatch.get(key);
      if (!list) { list = []; ownerByBatch.set(key, list); }
      list.push(ownerByPart[index]!);
    });
    return { batches: batchParts(parts), ownerByBatch };
  }, [buildings, now]);

  return <>{batches.map(batch => <InstanceBatch
    key={batch.key}
    batch={batch}
    night={night}
    owners={ownerByBatch.get(batch.key)}
    onSelectBuilding={onSelectBuilding}
  />)}</>;
}

const PlantsLayer = memo(function PlantsLayer({ night }: { night: boolean }) {
  const plants = useWorldStore(selectPlants);
  const batches = useMemo(() => {
    const parts: Part[] = [];
    for (const item of plants) {
      const at = cellPoint(item.gridPosition.x, item.gridPosition.z);
      const seed = hashId(item.id);
      const speciesScale = item.species === "tree" ? 2.2 : item.species === "shrub" ? 1.6 : 1.35;
      const growth = Math.max(0, Math.min(3, item.growth));

    const natureMultiplier =
      item.species === "tree" ? 1.65 :
      item.species === "shrub" ? 1.4 :
      item.species === "fern" ? 1.35 :
      item.species === "flower" ? 1.5 :
      1.25;

    const individualSize = 0.7 + (seed % 100) / 100 * 0.9;

    const scale =
      (0.25 + growth * 0.65) *
      individualSize *
      (CELL_SIZE / 1.5) *
      speciesScale *
      natureMultiplier;
      const root = compose([at.x, 0, at.z], [0, seed % 360 * Math.PI / 180, 0], [scale, scale, scale]);
      placeModel(plantModel(item.species, seed), root, parts);
    }
    return batchParts(parts);
  }, [plants]);
  return <BatchLayer batches={batches} night={night} />;
});

// Mirrors the displayed building floor calculation used by the renderer.
function displayedFloors(building: Building, now: number): number {
  const config = BUILDING_CATALOG.find(item => item.id === building.variantId);
  const stage = growthStage(buildingProgress(building, now));
  const fallback = 1 + ((config?.floors ?? 1) - 1) * stage / 2;
  return Math.min(MAX_BUILDING_FLOORS, Math.max(1, Math.round(building.floors ?? fallback)));
}

// The city's statistics reset at local 4 AM; unlike a fixed 24h calculation,
// calculating consecutive local 4 AM boundaries also handles DST changes.
function cityDayPercentage(timestamp: number): number {
  const start = new Date(timestamp);
  start.setHours(4, 0, 0, 0);
  if (timestamp < start.getTime()) start.setDate(start.getDate() - 1);

  const end = new Date(start);
  end.setDate(end.getDate() + 1);
  const duration = Math.max(1, end.getTime() - start.getTime());
  return Math.max(0, Math.min(100, (timestamp - start.getTime()) / duration * 100));
}

function piePoint(degrees: number, radius: number) {
  const radians = degrees * Math.PI / 180;
  return {
    x: PIE_CENTER + Math.cos(radians) * radius,
    y: PIE_CENTER + Math.sin(radians) * radius,
  };
}

function piePath(startDegrees: number, sweepDegrees: number) {
  const first = piePoint(startDegrees, PIE_RADIUS);
  const last = piePoint(startDegrees + sweepDegrees, PIE_RADIUS);
  const largeArc = sweepDegrees > 180 ? 1 : 0;
  return `M ${PIE_CENTER} ${PIE_CENTER} L ${first.x} ${first.y} A ${PIE_RADIUS} ${PIE_RADIUS} 0 ${largeArc} 1 ${last.x} ${last.y} Z`;

// Reflect a bubble off the inner glass boundaries.
// Reflection preserves movement rather than freezing at the wall.
function reflectWithinBounds(
  value: number,
  halfExtent: number,
  radius: number
): number {
  const limit = Math.max(0, halfExtent - radius - GLASS_CLEARANCE);
  if (limit === 0) return 0;

  const span = limit * 2;
  const period = span * 2;
  const wrapped = ((value + limit) % period + period) % period;

  return wrapped <= span
    ? -limit + wrapped
    : limit - (wrapped - span);
}

function BubbleMesh({ bubble }: { bubble: Bubble }) {
  const group = useRef<Group>(null);
  const shell = useRef<Group>(null);
  const film = useRef<Mesh>(null);
  const sparks = useRef<InstancedMesh>(null);
  const sparkMaterial = useRef<MeshBasicMaterial>(null);
  const clock = useThree(state => state.clock);
  const recordBubblePop = useWorldStore(state => state.recordBubblePop);
  const matrix = useMemo(() => new Matrix4(), []);
  const timeOffset = useRef(0);
  const popped = useRef<{ started: number; scale: number } | null>(null);

  function popBubble(event: ThreeEvent<MouseEvent>) {
    event.stopPropagation();
    const object = group.current;

    // Camera drags and right clicks should not pop bubbles.
    if (event.button !== 0 || event.delta > 5 || !object?.visible || popped.current) return;

    popped.current = {
      started: clock.getElapsedTime(),
      scale: object.scale.x,
    };

    recordBubblePop();
    if (shell.current) shell.current.visible = false;
  }

  useFrame(({ clock }) => {
    const object = group.current;
    if (!object) return;

    const t = clock.getElapsedTime();
    const pop = popped.current;

    if (pop) {
      const age = t - pop.started;

      if (age < RESPAWN_DELAY) {
        const progress = Math.min(1, age / POP_DURATION);
        object.visible = age < POP_DURATION;
        object.scale.setScalar(pop.scale);

        if (shell.current) shell.current.visible = false;

        if (sparks.current) {
          sparks.current.visible = age < POP_DURATION;

          const distance = bubble.radius * (0.65 + progress * 1.55);
          const size = bubble.radius * 0.095 * (1 - progress) + 0.002;

          SPARK_DIRECTIONS.forEach(([x, y, z], index) => {
            matrix.makeScale(size, size, size).setPosition(
              x * distance,
              y * distance,
              z * distance
            );
            sparks.current!.setMatrixAt(index, matrix);
          });

          sparks.current.instanceMatrix.needsUpdate = true;
        }

        if (sparkMaterial.current) {
          sparkMaterial.current.opacity = 0.9 * (1 - progress);
        }

        return;
      }

      // Restart at the bottom of the path with the existing fade-in.
      popped.current = null;
      timeOffset.current = t + bubble.phase - bubble.delay;
    }

    if (shell.current) shell.current.visible = true;
    if (sparks.current) sparks.current.visible = false;

    const elapsed = t + bubble.phase - bubble.delay - timeOffset.current;
    const cycle = ((elapsed % bubble.lifetime) + bubble.lifetime) % bubble.lifetime;

    // Each cycle begins near the bottom and travels up through the sky.
    const progress = cycle / bubble.lifetime;

    // Continuous X/Z drift plus overlapping waves creates a curved path.
    const x =
      bubble.x +
      bubble.driftX * cycle +
      Math.sin(cycle * 0.9 + bubble.phase) * bubble.sway +
      Math.sin(cycle * 0.37 + bubble.phase * 1.8) * bubble.sway * 0.45;

    const z =
      bubble.z +
      bubble.driftZ * cycle +
      Math.cos(cycle * 0.72 + bubble.phase) * bubble.sway * 0.65 +
      Math.sin(cycle * 0.43 + bubble.phase) * bubble.sway * 0.35;

    const y =
      MIN_HEIGHT +
      bubble.y +
      cycle * bubble.speed +
      Math.sin(cycle * 0.65 + bubble.phase) * 0.3;

    const safeX = reflectWithinBounds(x, HALF_TANK, bubble.radius);
    const safeZ = reflectWithinBounds(z, HALF_TANK, bubble.radius);

    object.position.set(safeX, y, safeZ);
    object.rotation.y = cycle * 0.16 + bubble.phase;
    object.rotation.z = Math.sin(cycle * 0.6 + bubble.phase) * 0.12;

    // Smoothly appear and disappear at the start/end of each cycle.
    const fadeIn = Math.min(1, progress / 0.12);
    const fadeOut = Math.min(1, (1 - progress) / 0.16);
    const opacity = Math.max(0, Math.min(fadeIn, fadeOut));

    object.scale.setScalar(Math.max(0.001, opacity));

    // Hide bubbles that have faded out.
    object.visible = opacity > 0.01;
  });

  const r = bubble.radius;

  return (
    <group ref={group} position={[bubble.x, MIN_HEIGHT + bubble.y, bubble.z]}>
      <group ref={shell}>
      {/* Iridescent, glossy soap-film shader */}

        ref={film}
        geometry={SPHERE}

        onClick={popBubble}
        raycast={(raycaster, intersections) => {
          const object = group.current;

        sizeRoll > 0.87
          ? 0.9 + random() * 0.65
          : 0.18 + random() * 0.52;

      const availableX = Math.max(

      );
      const availableZ = Math.max(
        0,
        HALF_TANK - radius - GLASS_CLEARANCE
      );

      const x =
        -availableX +
        ((column + 0.15 + random() * 0.7) / columns) *

        -availableZ +
        ((row + 0.15 + random() * 0.7) / rows) *

        x,
        y: random() * HEIGHT_RANGE,

import { useFrame } from "@react-three/fiber";
import { memo, useRef } from "react";
import { DoubleSide, ShaderMaterial } from "three";

type WaterKind = "river" | "lake" | "pond";

const vertexShader = `
uniform float uTime;
uniform float uKind;

varying vec3 vWorldPosition;
varying vec2 vUv;
varying float vSurfaceSlope;

void main() {
  vec3 displaced = position;
  vec2 p = position.xz;

  float river = 1.0 - step(0.5, uKind);
  float speed = mix(0.32, 1.4, river);

  // Broad, smooth undulations.
  float wave1 = sin(
    p.y * 2.1 -
    uTime * speed +
    sin(p.x * 0.65) * 0.4
  );

  float wave2 = sin(
    p.y * 3.8 -
    uTime * speed * 1.15 +
    p.x * 1.1
  );

  // Subtle rounded divots, not sharp geometric shapes.
  float depression = sin(
    p.x * 1.4 +
    p.y * 1.8 -
    uTime * speed * 0.75
  );

  depression = -pow(
    max(0.0, depression),
    2.0
  );

  float amplitude = mix(0.025, 0.055, river);

  float displacement =
    wave1 * amplitude * 0.55 +
    wave2 * amplitude * 0.22 +
    depression * amplitude * 0.38;

  displaced.y += displacement;

  // Slight shading variation for surface depth.
  vSurfaceSlope = displacement / max(amplitude, 0.001);

  vUv = position.xz;

  vec4 worldPosition =
    modelMatrix * vec4(displaced, 1.0);

  vWorldPosition = worldPosition.xyz;

  gl_Position =
    projectionMatrix * viewMatrix * worldPosition;
}
`;

const fragmentShader = `
uniform float uTime;
uniform float uKind;

varying vec3 vWorldPosition;
varying vec2 vUv;
varying float vSurfaceSlope;

void main() {
  float t = uTime;

  // River moves faster; pond moves slowly.
  float isRiver = 1.0 - step(0.5, uKind);
  float speed = mix(0.18, 1.15, isRiver);

  vec2 p = vUv;

  // Smooth, broad movement instead of sharp shapes.
  float wave1 = sin(
    p.y * 3.2 -
    t * speed +
    sin(p.x * 0.8) * 0.35
  );

  float wave2 = sin(
    p.y * 5.5 -
    t * speed * 1.25 +
    p.x * 0.65
  );

  float wave3 = sin(
    p.x * 2.0 +
    p.y * 1.4 -
    t * speed * 0.55
  );

  // Low contrast keeps the surface smooth.
  float movement =
    wave1 * 0.055 +
    wave2 * 0.025 +
    wave3 * 0.015;

  vec3 deepWater = vec3(0.035, 0.49, 0.66);
  vec3 clearWater = vec3(0.055, 0.75, 0.83);

  float blend = clamp(
    0.58 + movement,
    0.0,
    1.0
  );

  vec3 color = mix(deepWater, clearWater, blend);

  // Very subtle moving sheen, without foam shapes.
  float sheen = sin(
    p.y * 4.0 -
    t * speed * 0.85 +
    p.x * 0.4
  );

  sheen = smoothstep(0.7, 1.0, sheen);

  color += vec3(0.12, 0.18, 0.19) * sheen * 0.11;

  // Gentle brightness variation across the surface.
  float softLight = sin(
    p.x * 0.35 +
    p.y * 0.45 +
    t * 0.12
  );

  color += vec3(0.015, 0.025, 0.025) * softLight;

  color += vec3(0.015, 0.025, 0.03) * vSurfaceSlope;

  gl_FragColor = vec4(color, 1.0);
}
`;

export default memo(function WaterSurface({
  kind
}: {
  kind: WaterKind;
}) {
  const material = useRef<ShaderMaterial>(null);

  useFrame(({ clock }) => {
    if (material.current) {
      material.current.uniforms.uTime.value =
        clock.getElapsedTime();
    }
  });

  return (
    <shaderMaterial
      ref={material}
      vertexShader={vertexShader}
      fragmentShader={fragmentShader}
      uniforms={{
        uTime: { value: 0 },
        uKind: {
          value:
            kind === "river" ? 0 :
            kind === "lake" ? 1 : 2
        }
      }}
      side={DoubleSide}
    />
  );
});

export const SIMULATION_SPEEDS = [1, 2, 4, 8, 16] as const;
export type SimulationSpeed = (typeof SIMULATION_SPEEDS)[number];

export function nextSimulationSpeed(current: number): SimulationSpeed {
  const index = SIMULATION_SPEEDS.indexOf(current as SimulationSpeed);
  return SIMULATION_SPEEDS[(index + 1) % SIMULATION_SPEEDS.length]!;
}

/** Convert real elapsed time into the amount of simulation time to advance. */
export function scaledElapsedSeconds(elapsedMilliseconds: number, speed: number): number {
  if (!Number.isFinite(elapsedMilliseconds) || elapsedMilliseconds <= 0) return 0;
  if (!Number.isFinite(speed) || speed <= 0) return 0;
  return elapsedMilliseconds / 1_000 * speed;
}

function createBuilding(world: WorldState, at: number): Building | null {
  if (world.buildings.length >= MAX_BUILDINGS) return null;

  const residents = residentOccupiedPlots(world.residents);
  const waterPlacement = createWaterPlacement(world.waterSeed ?? 1);

  const catalog: (typeof BUILDING_CATALOG)[number][] = [];

  for (const entry of BUILDING_CATALOG) {
    const copies = entry.id === "solar-house" ? 3 : 2;

    for (let copy = 0; copy < copies; copy++) {
      catalog.push(entry);
    }
  }

  // Shuffle the weighted catalog. Every building type gets
  // a chance, including smaller buildings that fit gaps.
  for (let i = catalog.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    const temp = catalog[i];
    catalog[i] = catalog[j];
    catalog[j] = temp;
  }

  const attempted = new Set<string>();

  for (const config of catalog) {
    if (attempted.has(config.id)) continue;
    attempted.add(config.id);
    const plot = findFreePlot(
      [...world.buildings, ...residents],
      world.plants,
      config.width,
      config.depth,
      waterPlacement.isBuildingPlotValid
    );

    if (!plot) continue;

    return {
      type: "building",
      id: createId("building"),
      name: config.name,
      variantId: config.id,
      gridPosition: plot,
      footprint: {
        width: config.width,
        depth: config.depth
      },
      rotation: 0,
      floors: Math.min(config.floors, MAX_BUILDING_FLOORS),
      moss: 0,
      createdAt: at
    };
  }

  return null;
}


function createPlant(world: WorldState, at: number): Plant | null {
  if (world.plants.length >= MAX_PLANTS) return null;
  // Randomized candidates keep free navigation corridors around structures.
  const residents = residentOccupiedPlots(world.residents);
  const waterPlacement = createWaterPlacement(world.waterSeed ?? 1);
  const species = (
    ["tree", "tree", "tree", "tree", "tree",
     "shrub", "shrub", "fern", "flower", "grass"] as const
  )[Math.floor(Math.random() * 10)]!;

  const position = findFreePlantSite(
    [...world.buildings, ...world.plants, ...residents],
    Math.random,
    waterPlacement.isPlantSiteValid
  , species);
  if (!position) return null;
  
  return { type: "plant", id: createId("plant"), species, variantId: species, gridPosition: position, growth: 0.2 + Math.random() * 0.35, createdAt: at };
}




function movePlantsOffWater(input: WorldState): WorldState {
  const waterPlacement = createWaterPlacement(input.waterSeed ?? 1);
  const plants = [...input.plants];
  let changed = false;

  for (let index = 0; index < plants.length; index++) {
    const plant = plants[index];

    if (
      waterPlacement.isPlantSiteValid(
        plant.gridPosition.x,
        plant.gridPosition.z
      )
    ) {
      continue;
    }

    const occupied = [
      ...input.buildings,
      ...residentOccupiedPlots(input.residents),
      ...plants.filter((_, i) => i !== index),
    ];

    const position = findFreePlantSite(
      occupied,
      Math.random,
      waterPlacement.isPlantSiteValid
    );

    if (!position) continue;

    plants[index] = {
      ...plant,
      gridPosition: position,
    };

    changed = true;
  }

  return changed ? { ...input, plants } : input;
}

function fillOpenLand(input: WorldState, at: number): WorldState {
  if (input.buildings.length === 0 && input.plants.length === 0) {
    return input;
  }

  let world = input;

  const targetBuildings = Math.min(MAX_BUILDINGS, 65);

  while (world.buildings.length < targetBuildings) {
    const building = createBuilding(world, at);
    if (!building) break;

    world = {
      ...world,
      buildings: [...world.buildings, building],
    };
  }

  const targetPlants = Math.min(
    MAX_PLANTS,
    Math.max(310, world.buildings.length * 10)
  );

  while (world.plants.length < targetPlants) {
    const plant = createPlant(world, at);
    if (!plant) break;

    world = {
      ...world,
      plants: [...world.plants, plant],
    };
  }

  
  // Relocate saved trees whose mature canopies intersect
  // nearby buildings, without changing building positions.
  const repairedPlants = [...world.plants];
  const structures = [...world.buildings, ...world.residents];

  for (let i = 0; i < repairedPlants.length; i++) {
    const plant = repairedPlants[i];
    if (plant.species !== "tree") continue;

    const tooClose = structures.some(item => {
      const bx = item.gridPosition.x;
      const bz = item.gridPosition.z;
      const bw = item.footprint?.width ?? 0.5;
      const bd = item.footprint?.depth ?? 0.5;
      const px = plant.gridPosition.x + 0.25;
      const pz = plant.gridPosition.z + 0.25;

      const dx = Math.max(bx - px, 0, px - (bx + bw));
      const dz = Math.max(bz - pz, 0, pz - (bz + bd));

      return Math.hypot(dx, dz) < 1.0;
    });

    if (!tooClose) continue;

    const others = repairedPlants.filter(
      (_, index) => index !== i
    );

    const position = findFreePlantSite(
      [...structures, ...others],
      Math.random,
      waterPlacement.isPlantSiteValid,
      "tree"
    );

    if (position) {
      repairedPlants[i] = {
        ...plant,
        gridPosition: position
      };
    }
  }

  world = { ...world, plants: repairedPlants };

  return world;
}

const EDEN_GROWTH_MAX = 3;

function edenRandom(id: string, salt = 0): number {
  let hash = 2166136261 ^ salt;
  for (let i = 0; i < id.length; i++) {
    hash = Math.imul(hash ^ id.charCodeAt(i), 16777619);
  }
  return (hash >>> 0) / 4294967295;
}

function growEden(world: WorldState, points: number): WorldState {
  if (points <= 0) return world;

  return {
    ...world,
    buildings: world.buildings.map(building => {
      const speed = 0.65 + edenRandom(building.id, 7) * 0.7;
      return {
        ...building,
        moss: Math.min(
          1,
          (building.moss ?? 0) + points * 0.025 * speed
        ),
      };
    }),
    plants: world.plants.map(plant => {
      const speed = 0.65 + edenRandom(plant.id, 13) * 0.7;
      const maxGrowth =
        1.4 + edenRandom(plant.id, 29) * 1.6;

      return {
        ...plant,
        growth: Math.min(
          Math.min(EDEN_GROWTH_MAX, maxGrowth),
          plant.growth + points * 0.02625 * speed
        ),
      };
    }),
  };
}

function applyMilestones(worldInput: WorldState, at: number): WorldState {
  let world = worldInput;
  const consumedMilestones = { ...world.progression.consumedMilestones };
  for (const category of ["productivity", "leisure", "idle"] as const) {
    const total = world.activity[ACTIVITY_KEY[category]];
    const earned = Math.floor(total / ACTIVITY_MILESTONE_POINTS);
    let consumed = consumedMilestones[category];
    while (consumed < earned) {
      let changed = false;
      if (category === "productivity") {
        const eligible = world.buildings
          .filter(building =>
            (building.floors ?? 1) < MAX_BUILDING_FLOORS
          )
          .sort((a, b) =>
            (a.floors ?? 1) - (b.floors ?? 1)
          );

        // Seven of every ten productivity milestones prioritize
        // construction; the remaining three prioritize height.
        const shouldBuild =
          world.buildings.length < MAX_BUILDINGS &&
          consumed % 10 < 7;

        if (shouldBuild) {
          const building = createBuilding(world, at);

          if (building) {
            world = {
              ...world,
              buildings: [...world.buildings, building],
            };
            changed = true;
          }
        }

        if (!changed && eligible.length > 0) {
          const building = eligible[0];

          world = {
            ...world,
            buildings: world.buildings.map(item =>
              item.id === building.id
                ? {
                    ...item,
                    floors: Math.min(
                      MAX_BUILDING_FLOORS,
                      (item.floors ?? 1) + 1
                    ),
                    growth: 1,
                  }
                : item
            ),
          };

          changed = true;
        }

        if (!changed) {
          const building = createBuilding(world, at);

          if (building) {
            world = {
              ...world,
              buildings: [...world.buildings, building],
            };
            changed = true;
          }
        }
      } else if (category === "leisure") {
        const buildingCount = world.buildings.length;
        const treeCount = world.plants.filter(
          plant => plant.species === "tree"
        ).length;

        // Roughly 1 tree per building, with some variation.
        const maxTrees = Math.min(
          MAX_PLANTS,
          Math.max(4, Math.ceil(buildingCount * 0.65))
        );

        // Limit total nature so buildings remain visually prominent.
        const maxNature = Math.min(
          MAX_PLANTS,
          Math.max(8, Math.ceil(buildingCount * 1.5))
        );

        if (world.plants.length < maxNature) {
          let plant = createPlant(world, at);

          if (plant) {
            if (
              plant.species === "tree" &&
              treeCount >= maxTrees
            ) {
              const alternatives = [
                "shrub",
                "fern",
                "flower",
                "grass",
              ] as const;

              const species =
                alternatives[
                  Math.floor(Math.random() * alternatives.length)
                ]!;

              plant = {
                ...plant,
                species,
                variantId: species,
              };
            }

            world = {
              ...world,
              plants: [...world.plants, plant],
            };

            changed = true;
          }
        }

        // Existing buildings continue developing moss and vines.
        if (
          earned >= 3 &&
          world.buildings.some(
            building => (building.moss ?? 0) < 1
          )
        ) {
          world = {
            ...world,
            buildings: world.buildings.map(building => ({
              ...building,
              moss: Math.min(
                1,
                Math.max(
                  building.moss ?? 0,
                  (earned - 2) / 4
                )
              ),
            })),
          };

          changed = true;
        }

      } else if (world.residents.length < MAX_RESIDENTS) {
        const cell = findOpenResidentCell(world.buildings, world.plants, world.residents.map(resident => resident.position));
        if (!cell) break;
        const building = world.buildings[Math.floor(Math.random() * Math.max(1, world.buildings.length))];
        const resident: Resident = { type: "resident", id: createId("resident"), name: `Resident ${world.residents.length + 1}`, position: residentWorldPosition(cell), activity: "walking", homeId: building?.id, route: [], createdAt: at };
        world = { ...world, residents: [...world.residents, resident] }; changed = true;
      }
      if (!changed) break;
      consumed++;
    }
    consumedMilestones[category] = consumed;
  }
  return { ...world, progression: { ...world.progression, consumedMilestones } };
}

export const useWorldStore = create<WorldStore>((set) => ({

  // --------------------------------------------------
  // System-wide idle monitoring
  // --------------------------------------------------

  systemIdle: false,
  lastNonIdleCategory: getNonIdleCategory(
    DEFAULT_WORLD_STATE.activity.category
  ),

  setSystemIdle: (idle) =>
    set((state) => {
      if (state.systemIdle === idle) {
        return state;
      }

      const previousCategory = getNonIdleCategory(
        state.world.activity.category,
        state.lastNonIdleCategory
      );

      return {
        systemIdle: idle,
        lastNonIdleCategory: previousCategory,
        world: {
          ...state.world,
          lastActiveAt: idle
            ? state.world.lastActiveAt
            : Date.now(),
          activity: {
            ...state.world.activity,
            category: idle ? "idle" : previousCategory,
          },
        },
      };
    }),

  // --------------------------------------------------
  // World persistence
  // --------------------------------------------------

  resetWorld: (at = Date.now()) =>
    set((state) => {
      const world = freshWorld(at);
      const lastNonIdleCategory = getNonIdleCategory(
        world.activity.category
      );

      if (state.systemIdle) {
        world.activity.category = "idle";
      }

      return { world, lastNonIdleCategory };
    }),

  loadWorld: (savedWorld) =>
    set((state) => {
      const world = fillOpenLand(
        movePlantsOffWater(structuredClone(savedWorld)),
        Date.now()
      );
      const lastNonIdleCategory = getNonIdleCategory(
        world.activity.category,
        state.lastNonIdleCategory
      );

      if (state.systemIdle) {
        world.activity.category = "idle";
      } else if (world.activity.category === "idle") {
        world.activity.category = lastNonIdleCategory;
      }

      return { world, lastNonIdleCategory };
    }),

  // --------------------------------------------------
  // Buildings
  // --------------------------------------------------

  addBuilding: (building) =>
    set((state) => ({
      world: {
        ...state.world,
        buildings: [
          ...state.world.buildings,
          structuredClone(building),
        ],
      },
    })),

  removeBuilding: (id) =>
    set((state) => ({
      world: {
        ...state.world,
        buildings: state.world.buildings.filter(
          (building) => building.id !== id
        ),
      },
    })),

  // --------------------------------------------------
  // Plants
  // --------------------------------------------------

  addPlant: (plant) =>
    set((state) => ({
      world: {
        ...state.world,
        plants: [
          ...state.world.plants,
          structuredClone(plant),
        ],
      },
    })),

  removePlant: (id) =>
    set((state) => ({
      world: {
        ...state.world,
        plants: state.world.plants.filter(
          (plant) => plant.id !== id
        ),
      },
    })),

  // --------------------------------------------------
  // Residents
  // --------------------------------------------------

  addResident: (resident) =>
    set((state) => ({
      world: {
        ...state.world,
        residents: [
          ...state.world.residents,
          structuredClone(resident),
        ],
      },
    })),

  removeResident: (id) =>
    set((state) => ({
      world: {
        ...state.world,
        residents: state.world.residents.filter(
          (resident) => resident.id !== id
        ),
      },
    })),

  // --------------------------------------------------
  // Activity category
  // --------------------------------------------------

  setActivityCategory: (category, at = Date.now()) =>
    set((state) => {
      const lastNonIdleCategory = getNonIdleCategory(
        category,
        state.lastNonIdleCategory
      );

      return {
        lastNonIdleCategory,
        world: {
          ...state.world,
          lastActiveAt: at,
          activity: {
            ...state.world.activity,
            category: state.systemIdle ? "idle" : category,
          },
        },
      };
    }),

  recordInput: (at = Date.now()) =>
    set((state) => ({
      world: {
        ...state.world,
        lastActiveAt: at,
        activity: {
          ...state.world.activity,
          category: state.systemIdle
            ? "idle"
            : getNonIdleCategory(
                state.world.activity.category,
                state.lastNonIdleCategory
              ),
        },
      },
    })),

  // A successful bubble click, separate from the activity pie categories.
  recordBubblePop: () =>
    set(state => ({
      world: {
        ...state.world,
        activity: {
          ...state.world.activity,
          bubblePops: (state.world.activity.bubblePops ?? 0) + 1,
        },
      },
    })),

  addActivityPoints: (category, points) => {
    if (!Number.isFinite(points) || points <= 0) return;

    set(state => {
      let world = applyMilestones({
        ...state.world,
        activity: {
          ...state.world.activity,
          [ACTIVITY_KEY[category]]:
            state.world.activity[ACTIVITY_KEY[category]] + points,
        },
      }, Date.now());

      world = growEden(world, points);

      return { world };
    });
  },

  // --------------------------------------------------
  // Progression
  // --------------------------------------------------

  unlockBuilding: (buildingType) =>
    set((state) => {
      if (
        state.world.progression.unlockedBuildings.includes(
          buildingType
        )
      ) {
        return state;
      }

      return {
        world: {
          ...state.world,
          progression: {
            ...state.world.progression,
            unlockedBuildings: [
              ...state.world.progression.unlockedBuildings,
              buildingType,
            ],
          },
        },
      };
    }),

  // --------------------------------------------------
  // Simulation timer
  // --------------------------------------------------

  advanceTime: (seconds, at = Date.now()) => {
    if (!Number.isFinite(seconds) || seconds <= 0) return;
    set(state => {
      const dayKey = getCityDayKey(new Date(at));
      let world = state.world;
      if (world.cityDayKey !== dayKey) {
        world = { ...freshWorld(at), weather: dailyWeather(dayKey, at) };
      }

      // Electron determines whether the system is idle.
      // Otherwise, use the selected productivity/leisure/idle category.
      const category: ActivityCategory = state.systemIdle ? "idle" : world.activity.category;

      world = { ...world, simulationTime: world.simulationTime + seconds, activity: { ...world.activity, category, [ACTIVITY_KEY[category]]: world.activity[ACTIVITY_KEY[category]] + seconds / 60 } };
      if (world.residents.length) {
        const isNight = world.simulationTime % 600 >= 360;
        world = { ...world, residents: world.residents.map(resident => {
          if (isNight) {
            const home = world.buildings.find(building => building.id === resident.homeId);
            const homePosition = home ? {
              x: (home.gridPosition.x + home.footprint.width / 2 - GRID_SIZE / 2) * 2,
              y: 0,
              z: (home.gridPosition.z + home.footprint.depth / 2 - GRID_SIZE / 2) * 2,
            } : resident.position;
            return { ...resident, activity: "resting", position: homePosition, route: [] };
          }
          let route = resident.route ?? [];
          if (!route.length) {
            const target = findOpenResidentCell(world.buildings, world.plants, world.residents.map(other => other.position));
            if (target) route = findResidentPath(residentGridPosition(resident.position), target, world.buildings, world.plants) ?? [];
          }
          const nextCell = route[0];
          if (!nextCell) return { ...resident, activity: "walking", route: [] };
          const target = residentWorldPosition(nextCell);
          const dx = target.x - resident.position.x, dz = target.z - resident.position.z, distance = Math.hypot(dx, dz);
          const step = Math.min(distance, Math.min(seconds, 2) * 0.35);
          if (distance <= step + 0.01) return { ...resident, activity: "walking", position: target, route: route.slice(1) };
          return { ...resident, activity: "walking", route, position: { ...resident.position, x: resident.position.x + dx / distance * step, z: resident.position.z + dz / distance * step } };
        }) };
      }

      world = growEden(applyMilestones(world, at), seconds / 60);

      return { world };
    });
  },

  // --------------------------------------------------
  // Weather
  // --------------------------------------------------

  setWeather: weather => set(state => ({
      world: {
        ...state.world,
        weather: structuredClone(weather),
      },
    })),
}));

export const selectWeather = (state: WorldStore) => state.world.weather;

import { describe, expect, it } from "vitest";
import { nextSimulationSpeed, scaledElapsedSeconds, SIMULATION_SPEEDS } from "../simulation/timeControls";

describe("simulation time controls", () => {
  it("cycles through supported time multipliers and wraps back to real time", () => {
    expect(SIMULATION_SPEEDS).toEqual([1, 2, 4, 8, 16]);
    expect(nextSimulationSpeed(1)).toBe(2);
    expect(nextSimulationSpeed(2)).toBe(4);
    expect(nextSimulationSpeed(4)).toBe(8);
    expect(nextSimulationSpeed(8)).toBe(16);
    expect(nextSimulationSpeed(16)).toBe(1);
    expect(nextSimulationSpeed(99)).toBe(1);
  });

  it("scales elapsed real seconds and rejects invalid inputs", () => {
    expect(scaledElapsedSeconds(1_000, 1)).toBe(1);
    expect(scaledElapsedSeconds(1_000, 4)).toBe(4);
    expect(scaledElapsedSeconds(250, 8)).toBe(2);
    expect(scaledElapsedSeconds(0, 4)).toBe(0);
    expect(scaledElapsedSeconds(-500, 2)).toBe(0);
    expect(scaledElapsedSeconds(Number.NaN, 2)).toBe(0);
    expect(scaledElapsedSeconds(1_000, Number.POSITIVE_INFINITY)).toBe(0);
  });
});
